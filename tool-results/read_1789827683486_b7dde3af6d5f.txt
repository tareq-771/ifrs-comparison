     1→"use client";
     2→
     3→import {
     4→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
     5→} from "@/components/ui/table";
     6→import { cn } from "@/lib/utils";
     7→import {
     8→  type Categorized, type ChangeStatus, type CompareMode, type MatchedRow, type RowLeafInfo, type Totals,
     9→  computeAvgCumulative, computeChangePct, computeChangeStatus, computeMonthAvgDiff, computeMonthAvgDiffPct, computeMonthRatio, computeMonthStatus, computeMonthToAvgRatio,
    10→  fmtAmount, fmtPct, leafInfo, rowAmount,
    11→} from "@/lib/accounts";
    12→
    13→interface ResultsTableProps {
    14→  cat: Categorized;
    15→  T: Totals;
    16→  L1: string;
    17→  L2: string;
    18→  netRow: MatchedRow | null;
    19→  compareMode: CompareMode;
    20→  numMonths: number;
    21→}
    22→
    23→type RenderRow =
    24→  | { kind: "section"; label: string }
    25→  | { kind: "data"; r: MatchedRow; info: RowLeafInfo; isRevenue: boolean }
    26→  | { kind: "sub"; label: string; v1: number; v2: number; isRevenue: boolean }
    27→  | { kind: "total"; label: string; v1: number; v2: number; isRevenue: boolean }
    28→  | { kind: "memo"; label: string; v1: number | null; v2: number | null; ok: boolean };
    29→
    30→export function ResultsTable({ cat, T, L1, L2, netRow, compareMode, numMonths }: ResultsTableProps) {
    31→  const b1 = T.rev1;
    32→  const b2 = T.rev2;
    33→  const isMonthMode = compareMode === "monthCumulative";
    34→
    35→  // Column labels depend on mode
    36→  const col2Header = isMonthMode ? "الشهر الحالي" : L2;
    37→  const col2Sub = isMonthMode ? "(الشهر)" : "(الحالية)";
    38→  const col1Header = isMonthMode ? "التراكمي للسنة" : L1;
    39→  const col1Sub = isMonthMode ? "(التراكمي)" : "(المقارنة)";
    40→
    41→  const sectionData = (rows: MatchedRow[]): { r: MatchedRow; info: RowLeafInfo }[] => {
    42→    const info = leafInfo(rows);
    43→    return rows.map((r, i) => ({ r, info: info[i] }));
    44→  };
    45→
    46→  const rows: RenderRow[] = [];
    47→
    48→  // Revenue
    49→  rows.push({ kind: "section", label: "الإيرادات — Revenue" });
    50→  for (const { r, info } of sectionData(cat.sales)) rows.push({ kind: "data", r, info, isRevenue: true });
    51→  rows.push({ kind: "sub", label: "صافي الإيرادات (المبيعات − المردود)", v1: T.rev1, v2: T.rev2, isRevenue: true });
    52→
    53→  // Cost of sales
    54→  rows.push({ kind: "section", label: "تكلفة الإيرادات — Cost of Sales" });
    55→  for (const { r, info } of sectionData(cat.cost)) rows.push({ kind: "data", r, info, isRevenue: false });
    56→  rows.push({ kind: "sub", label: "صافي تكلفة الإيرادات", v1: T.cost1, v2: T.cost2, isRevenue: false });
    57→
    58→  // Gross profit
    59→  rows.push({ kind: "total", label: "مجمل الربح (الخسارة) — Gross Profit", v1: T.gross1, v2: T.gross2, isRevenue: true });
    60→
    61→  // Operating expenses (function of expense)
    62→  rows.push({ kind: "section", label: "مصاريف التشغيل — Operating Expenses" });
    63→
    64→  rows.push({ kind: "section", label: "مصاريف البيع والتوزيع" });
    65→  for (const { r, info } of sectionData(cat.sell)) rows.push({ kind: "data", r, info, isRevenue: false });
    66→  rows.push({ kind: "sub", label: "إجمالي البيع والتوزيع", v1: T.sell1, v2: T.sell2, isRevenue: false });
    67→
    68→  rows.push({ kind: "section", label: "المصاريف الإدارية والعمومية" });
    69→  for (const { r, info } of sectionData(cat.adm)) rows.push({ kind: "data", r, info, isRevenue: false });
    70→  rows.push({ kind: "sub", label: "إجمالي الإدارية والعمومية", v1: T.adm1, v2: T.adm2, isRevenue: false });
    71→
    72→  rows.push({ kind: "section", label: "مصاريف تشغيلية أخرى" });
    73→  for (const { r, info } of sectionData(cat.oth)) rows.push({ kind: "data", r, info, isRevenue: false });
    74→  rows.push({ kind: "sub", label: "إجمالي التشغيلية الأخرى", v1: T.oth1, v2: T.oth2, isRevenue: false });
    75→
    76→  rows.push({ kind: "sub", label: "إجمالي مصاريف التشغيل", v1: T.opex1, v2: T.opex2, isRevenue: false });
    77→  rows.push({ kind: "total", label: "الربح (الخسارة) من العمليات التشغيلية — Operating Profit", v1: T.oper1, v2: T.oper2, isRevenue: true });
    78→
    79→  // Other income
    80→  if (cat.othRev.length) {
    81→    rows.push({ kind: "section", label: "الدخل من مصادر أخرى — Other Income" });
    82→    for (const { r, info } of sectionData(cat.othRev)) rows.push({ kind: "data", r, info, isRevenue: true });
    83→    rows.push({ kind: "sub", label: "إجمالي الدخل من مصادر أخرى", v1: T.othRev1, v2: T.othRev2, isRevenue: true });
    84→  }
    85→
    86→  // Finance & tax
    87→  if (cat.fin.length) {
    88→    rows.push({ kind: "section", label: "تكاليف التمويل — Finance Costs" });
    89→    for (const { r, info } of sectionData(cat.fin)) rows.push({ kind: "data", r, info, isRevenue: false });
    90→    rows.push({ kind: "sub", label: "إجمالي تكاليف التمويل", v1: T.fin1, v2: T.fin2, isRevenue: false });
    91→  }
    92→  if (cat.fin.length || cat.tax.length) {
    93→    rows.push({ kind: "sub", label: "الربح (الخسارة) قبل الضريبة", v1: T.pbt1, v2: T.pbt2, isRevenue: true });
    94→  }
    95→  if (cat.tax.length) {
    96→    rows.push({ kind: "section", label: "ضريبة الدخل — Income Tax" });
    97→    for (const { r, info } of sectionData(cat.tax)) rows.push({ kind: "data", r, info, isRevenue: false });
    98→    rows.push({ kind: "sub", label: "إجمالي ضريبة الدخل", v1: T.tax1, v2: T.tax2, isRevenue: false });
    99→  }
   100→
   101→  // Net profit
   102→  rows.push({ kind: "total", label: "صافي الربح (الخسارة) للفترة — Net Profit", v1: T.net1, v2: T.net2, isRevenue: true });
   103→
   104→  // OCI
   105→  if (cat.oci.length) {
   106→    rows.push({ kind: "section", label: "الدخل الشامل الآخر — Other Comprehensive Income" });
   107→    for (const { r, info } of sectionData(cat.oci)) rows.push({ kind: "data", r, info, isRevenue: true });
   108→    rows.push({ kind: "sub", label: "إجمالي الدخل الشامل الآخر", v1: T.oci1, v2: T.oci2, isRevenue: true });
   109→    rows.push({ kind: "total", label: "إجمالي الدخل الشامل للفترة — Total Comprehensive Income", v1: T.comp1, v2: T.comp2, isRevenue: true });
   110→  }
   111→
   112→  // Reconciliation
   113→  if (netRow) {
   114→    const n1 = netRow.a1 ? Math.round((netRow.a1.d - netRow.a1.m) * 100) / 100 : null;
   115→    const n2 = netRow.a2 ? Math.round((netRow.a2.d - netRow.a2.m) * 100) / 100 : null;
   116→    const ok = (n1 != null && Math.abs(n1 - T.net1) < 0.05) && (n2 != null && Math.abs(n2 - T.net2) < 0.05);
   117→    rows.push({ kind: "memo", label: "التسوية: حساب صافي الدخل في دفتر الأستاذ", v1: n1, v2: n2, ok });
   118→  }
   119→
   120→  const pctOf = (v: number, base: number) => (base !== 0 ? v / base : null);
   121→
   122→  const statusClass = (st: MatchedRow["st"]): string => {
   123→    if (st === "مطابق") return "bg-emerald-50/60 dark:bg-emerald-950/20";
   124→    if (st === "في الملف الأول فقط") return "bg-amber-50/60 dark:bg-amber-950/20";
   125→    return "bg-orange-50/50 dark:bg-orange-950/20";
   126→  };
   127→
   128→  const changeStatusClass = (cs: ChangeStatus): string => {
   129→    if (cs === "مرغوب نمو" || cs === "مرغوب وفرة") return "text-emerald-600 dark:text-emerald-400";
   130→    if (cs === "غير مرغوب انخفاض" || cs === "غير مرغوب زيادة") return "text-rose-600 dark:text-rose-400";
   131→    return "text-slate-400";
   132→  };
   133→
   134→  return (
   135→    <div className="scroll-thin max-h-[65vh] overflow-auto rounded-xl border border-slate-200 bg-white shadow-sm dark:border-slate-800 dark:bg-slate-950">
   136→      <Table className="text-xs">
   137→        <TableHeader className="sticky top-0 z-10">
   138→          <TableRow className="border-slate-200 bg-slate-800 hover:bg-slate-800 dark:border-slate-700 dark:bg-slate-900">
   139→            <TableHead className="h-9 w-12 text-center text-slate-50">رقم</TableHead>
   140→            <TableHead className="h-9 text-slate-50">البند</TableHead>
   141→            {isMonthMode ? (
   142→              <>
   143→                <TableHead className="h-9 text-center text-slate-50">{col2Header}<br /><span className="font-normal opacity-80">{col2Sub}</span></TableHead>
   144→                <TableHead className="h-9 text-center text-slate-50">نسبة الشهر<br /><span className="font-normal opacity-80">من التراكمي</span></TableHead>
   145→                <TableHead className="h-9 text-center text-slate-50">نسبة البند<br /><span className="font-normal opacity-80">لمبيعات الشهر</span></TableHead>
   146→                <TableHead className="h-9 text-center text-slate-50">{col1Header}<br /><span className="font-normal opacity-80">{col1Sub}</span></TableHead>
   147→                <TableHead className="h-9 text-center text-slate-50">% من<br /><span className="font-normal opacity-80">الإيرادات</span></TableHead>
   148→                <TableHead className="h-9 text-center text-slate-50">المتوسط<br /><span className="font-normal opacity-80">للتراكمي</span></TableHead>
   149→                <TableHead className="h-9 text-center text-slate-50">نسبة الشهر<br /><span className="font-normal opacity-80">من المتوسط</span></TableHead>
   150→                <TableHead className="h-9 text-center text-slate-50">فرق الشهر<br /><span className="font-normal opacity-80">عن المتوسط</span></TableHead>
   151→                <TableHead className="h-9 text-center text-slate-50">نسبة الفرق<br /><span className="font-normal opacity-80">عن المتوسط</span></TableHead>
   152→                <TableHead className="h-9 text-center text-slate-50">حالة الشهر<br /><span className="font-normal opacity-80">مقابل المتوسط</span></TableHead>
   153→              </>
   154→            ) : (
   155→              <>
   156→                <TableHead className="h-9 text-center text-slate-50">{col2Header}<br /><span className="font-normal opacity-80">{col2Sub}</span></TableHead>
   157→                <TableHead className="h-9 text-center text-slate-50">% من<br /><span className="font-normal opacity-80">الإيرادات</span></TableHead>
   158→                <TableHead className="h-9 text-center text-slate-50">{col1Header}<br /><span className="font-normal opacity-80">{col1Sub}</span></TableHead>
   159→                <TableHead className="h-9 text-center text-slate-50">% من<br /><span className="font-normal opacity-80">الإيرادات</span></TableHead>
   160→                <TableHead className="h-9 text-center text-slate-50">التغير</TableHead>
   161→                <TableHead className="h-9 text-center text-slate-50">نسبة<br /><span className="font-normal opacity-80">التغير</span></TableHead>
   162→                <TableHead className="h-9 text-center text-slate-50">حالة<br /><span className="font-normal opacity-80">التغير</span></TableHead>
   163→              </>
   164→            )}
   165→            <TableHead className="h-9 text-center text-slate-50">الحالة</TableHead>
   166→          </TableRow>
   167→        </TableHeader>
   168→        <TableBody>
   169→          {rows.map((row, idx) => {
   170→            if (row.kind === "section") {
   171→              const isMain = row.label.includes("—");
   172→              return (
   173→                <TableRow
   174→                  key={`sec-${idx}`}
   175→                  className={cn(
   176→                    "border-slate-200 dark:border-slate-800",
   177→                    isMain ? "bg-slate-100 dark:bg-slate-900/60" : "bg-slate-50 dark:bg-slate-900/30"
   178→                  )}
   179→                >
   180→                  <TableCell colSpan={isMonthMode ? 13 : 10} className={cn("py-1.5", isMain ? "font-bold text-slate-800 dark:text-slate-100" : "font-semibold text-slate-600 dark:text-slate-300")}>
   181→                    {row.label}
   182→                  </TableCell>
   183→                </TableRow>
   184→              );
   185→            }
   186→
   187→            if (row.kind === "data") {
   188→              const { r, info, isRevenue } = row;
   189→              const a = r.a1 ?? r.a2!;
   190→              const v1 = rowAmount(r, "a1", isRevenue);
   191→              const v2 = rowAmount(r, "a2", isRevenue);
   192→              const isParent = (r.a1 && !info.leaf1) || (r.a2 && !info.leaf2);
   193→
   194→              if (isMonthMode) {
   195→                // Month/cumulative mode columns (13 total):
   196→                // رقم | البند | الشهر(v2) | نسبة الشهر من التراكمي(v2/v1) | نسبة البند لمبيعات الشهر(v2/b2) | التراكمي(v1) | % من الإيرادات(v1/b1) | المتوسط للتراكمي(v1/nm) | نسبة الشهر من المتوسط(v2/avg) | فرق الشهر عن المتوسط(v2-avg) | نسبة الفرق عن المتوسط((v2-avg)/avg) | حالة الشهر | الحالة
   197→                const monthRatio = computeMonthRatio(v2, v1);
   198→                const itemToMonthSales = b2 !== 0 && v2 != null ? v2 / b2 : null;
   199→                const s1 = b1 !== 0 && v1 != null ? v1 / b1 : null;
   200→                const avgCumulative = computeAvgCumulative(v1, numMonths);
   201→                const monthToAvg = computeMonthToAvgRatio(v2, avgCumulative);
   202→                const avgDiff = computeMonthAvgDiff(v2, avgCumulative);
   203→                const avgDiffPct = computeMonthAvgDiffPct(v2, avgCumulative);
   204→                const cs = computeMonthStatus(monthRatio, isRevenue, numMonths);
   205→                return (
   206→                  <TableRow
   207→                    key={`d-${idx}`}
   208→                    className={cn("border-slate-100 transition-colors dark:border-slate-800/60", statusClass(r.st), isParent && "font-bold")}
   209→                  >
   210→                    <TableCell className="text-center text-slate-500 dark:text-slate-400">{a.num ?? "—"}</TableCell>
   211→                    <TableCell className={cn("text-slate-800 dark:text-slate-100", isParent && "font-bold")}>{a.nm}</TableCell>
   212→                    <TableCell className={cn("tnum text-center", v2 != null && v2 < 0 ? "text-rose-600 dark:text-rose-400" : "text-slate-700 dark:text-slate-200", isParent && "font-bold")}>{fmtAmount(v2)}</TableCell>
   213→                    <TableCell className="tnum text-center font-semibold text-emerald-600 dark:text-emerald-400">{fmtPct(monthRatio)}</TableCell>
   214→                    <TableCell className="tnum text-center text-slate-500 dark:text-slate-400">{fmtPct(itemToMonthSales)}</TableCell>
   215→                    <TableCell className={cn("tnum text-center", v1 != null && v1 < 0 ? "text-rose-600 dark:text-rose-400" : "text-slate-700 dark:text-slate-200", isParent && "font-bold")}>{fmtAmount(v1)}</TableCell>
   216→                    <TableCell className="tnum text-center text-slate-500 dark:text-slate-400">{fmtPct(s1)}</TableCell>
   217→                    <TableCell className="tnum text-center text-slate-600 dark:text-slate-300">{fmtAmount(avgCumulative)}</TableCell>
   218→                    <TableCell className={cn("tnum text-center font-semibold", monthToAvg != null && monthToAvg > 1 ? "text-emerald-600 dark:text-emerald-400" : monthToAvg != null && monthToAvg < 1 ? "text-rose-600 dark:text-rose-400" : "text-slate-400")}>{fmtPct(monthToAvg)}</TableCell>
   219→                    <TableCell className={cn("tnum text-center font-medium", avgDiff != null && avgDiff < 0 ? "text-rose-600 dark:text-rose-400" : "text-emerald-600 dark:text-emerald-400")}>{fmtAmount(avgDiff)}</TableCell>
   220→                    <TableCell className={cn("tnum text-center font-medium", avgDiffPct != null && avgDiffPct < 0 ? "text-rose-600 dark:text-rose-400" : "text-emerald-600 dark:text-emerald-400")}>{fmtPct(avgDiffPct)}</TableCell>
   221→                    <TableCell className={cn("text-center font-semibold", changeStatusClass(cs))}>{cs}</TableCell>
   222→                    <TableCell className="text-center text-slate-500 dark:text-slate-400">{r.st}</TableCell>
   223→                  </TableRow>
   224→                );
   225→              }
   226→
   227→              // Period comparison mode (original)
   228→              const s1 = b2 !== 0 && v2 != null ? v2 / b2 : null;
   229→              const s2 = b1 !== 0 && v1 != null ? v1 / b1 : null;
   230→              const chg = v1 != null && v2 != null ? Math.round((v2 - v1) * 100) / 100 : null;
   231→              const cp = computeChangePct(chg, v1);
   232→              const cs = computeChangeStatus(chg, isRevenue);
   233→              return (
   234→                <TableRow
   235→                  key={`d-${idx}`}
   236→                  className={cn("border-slate-100 transition-colors dark:border-slate-800/60", statusClass(r.st), isParent && "font-bold")}
   237→                >
   238→                  <TableCell className="text-center text-slate-500 dark:text-slate-400">{a.num ?? "—"}</TableCell>
   239→                  <TableCell className={cn("text-slate-800 dark:text-slate-100", isParent && "font-bold")}>{a.nm}</TableCell>
   240→                  <TableCell className={cn("tnum text-center", v2 != null && v2 < 0 ? "text-rose-600 dark:text-rose-400" : "text-slate-700 dark:text-slate-200", isParent && "font-bold")}>{fmtAmount(v2)}</TableCell>
   241→                  <TableCell className="tnum text-center text-slate-500 dark:text-slate-400">{fmtPct(s1)}</TableCell>
   242→                  <TableCell className={cn("tnum text-center", v1 != null && v1 < 0 ? "text-rose-600 dark:text-rose-400" : "text-slate-700 dark:text-slate-200", isParent && "font-bold")}>{fmtAmount(v1)}</TableCell>
   243→                  <TableCell className="tnum text-center text-slate-500 dark:text-slate-400">{fmtPct(s2)}</TableCell>
   244→                  <TableCell className={cn("tnum text-center font-medium", chg != null && chg < 0 ? "text-rose-600 dark:text-rose-400" : "text-slate-600 dark:text-slate-300")}>{fmtAmount(chg)}</TableCell>
   245→                  <TableCell className="tnum text-center text-slate-500 dark:text-slate-400">{fmtPct(cp)}</TableCell>
   246→                  <TableCell className={cn("text-center font-semibold", changeStatusClass(cs))}>{cs}</TableCell>
   247→                  <TableCell className="text-center text-slate-500 dark:text-slate-400">{r.st}</TableCell>
   248→                </TableRow>
   249→              );
   250→            }
   251→
   252→            if (row.kind === "memo") {
   253→              if (isMonthMode) {
   254→                const monthRatio = computeMonthRatio(row.v2, row.v1);
   255→                const itemToMonthSales = b2 !== 0 && row.v2 != null ? row.v2 / b2 : null;
   256→                const s1 = b1 !== 0 && row.v1 != null ? row.v1 / b1 : null;
   257→                const avgCumulative = computeAvgCumulative(row.v1, numMonths);
   258→                const monthToAvg = computeMonthToAvgRatio(row.v2, avgCumulative);
   259→                const avgDiff = computeMonthAvgDiff(row.v2, avgCumulative);
   260→                const avgDiffPct = computeMonthAvgDiffPct(row.v2, avgCumulative);
   261→                const cs = computeMonthStatus(monthRatio, true, numMonths);
   262→                return (
   263→                  <TableRow key={`memo-${idx}`} className="border-slate-200 bg-amber-50 dark:border-slate-800 dark:bg-amber-950/20">
   264→                    <TableCell className="text-center text-slate-400">—</TableCell>
   265→                    <TableCell className="font-bold text-slate-700 dark:text-slate-200">{row.label}</TableCell>
   266→                    <TableCell className="tnum text-center font-bold text-slate-700 dark:text-slate-200">{fmtAmount(row.v2)}</TableCell>
   267→                    <TableCell className="tnum text-center font-bold text-emerald-600 dark:text-emerald-400">{fmtPct(monthRatio)}</TableCell>
   268→                    <TableCell className="tnum text-center text-slate-500 dark:text-slate-400">{fmtPct(itemToMonthSales)}</TableCell>
   269→                    <TableCell className="tnum text-center font-bold text-slate-700 dark:text-slate-200">{fmtAmount(row.v1)}</TableCell>
   270→                    <TableCell className="tnum text-center text-slate-500 dark:text-slate-400">{fmtPct(s1)}</TableCell>
   271→                    <TableCell className="tnum text-center font-bold text-slate-600 dark:text-slate-300">{fmtAmount(avgCumulative)}</TableCell>
   272→                    <TableCell className="tnum text-center font-bold text-slate-600 dark:text-slate-300">{fmtPct(monthToAvg)}</TableCell>
   273→                    <TableCell className={cn("tnum text-center font-bold", avgDiff != null && avgDiff < 0 ? "text-rose-600 dark:text-rose-400" : "text-emerald-600 dark:text-emerald-400")}>{fmtAmount(avgDiff)}</TableCell>
   274→                    <TableCell className={cn("tnum text-center font-bold", avgDiffPct != null && avgDiffPct < 0 ? "text-rose-600 dark:text-rose-400" : "text-emerald-600 dark:text-emerald-400")}>{fmtPct(avgDiffPct)}</TableCell>
   275→                    <TableCell className={cn("text-center font-semibold", changeStatusClass(cs))}>{cs}</TableCell>
   276→                    <TableCell className={cn("text-center font-bold", row.ok ? "text-emerald-600 dark:text-emerald-400" : "text-amber-600 dark:text-amber-400")}>{row.ok ? "مطابق ✓" : "⚠ فرق"}</TableCell>
   277→                  </TableRow>
   278→                );
   279→              }
   280→              const chg = row.v1 != null && row.v2 != null ? Math.round((row.v2 - row.v1) * 100) / 100 : null;
   281→              const cp = computeChangePct(chg, row.v1);
   282→              const cs = computeChangeStatus(chg, true);
   283→              return (
   284→                <TableRow key={`memo-${idx}`} className="border-slate-200 bg-amber-50 dark:border-slate-800 dark:bg-amber-950/20">
