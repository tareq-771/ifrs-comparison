    45→function ColumnSelect({ headers, value, onChange, placeholder }: {
    46→  headers: string[]; value: number | null; onChange: (v: number) => void; placeholder: string;
    47→}) {
    48→  return (
    49→    <Select value={value == null ? "" : String(value)} onValueChange={(v) => onChange(Number(v))} disabled={headers.length === 0}>
    50→      <SelectTrigger className="w-full" size="sm"><SelectValue placeholder={placeholder} /></SelectTrigger>
    51→      <SelectContent className="max-h-72">
    52→        {headers.map((h, j) => (
    53→          <SelectItem key={j} value={String(j)}>
    54→            <span className="text-muted-foreground">{j}</span>{" — "}<span className="truncate">{h || `(فارغ ${j})`}</span>
    55→          </SelectItem>
    56→        ))}
    57→      </SelectContent>
    58→    </Select>
    59→  );
    60→}

   408→  async function handleCreateGroup() {
   409→    const name = newGroupName.trim();
   410→    if (!name) {
   411→      toast({ title: "اسم المجموعة مطلوب", variant: "destructive" });
   412→      return;
   413→    }
   414→    // Case-insensitive duplicate check (client-side, before hitting the server)
   415→    const dup = groups.some((g) => g.name.trim().toLowerCase() === name.toLowerCase());
   416→    if (dup) {
   417→      toast({ title: "يوجد مجموعة بنفس الاسم", variant: "destructive" });
   418→      return;
   419→    }
   420→    setCreatingGroup(true);

   422→      const res = await fetch("/api/groups", {
   423→        method: "POST",
   424→        headers: { "Content-Type": "application/json" },
   425→        body: JSON.stringify({ name }),
   426→      });
   427→      const data = await res.json().catch(() => ({}));
   428→      if (!res.ok) throw new Error((data as { error?: string }).error || "فشل الإنشاء");
   429→      toast({ title: "تم إنشاء المجموعة", description: name });
   430→      setNewGroupName("");
   431→      // Refresh groups
   432→      const gRes = await fetch("/api/groups", { cache: "no-store" });
   433→      if (gRes.ok) setGroups(await gRes.json());
   434→    } catch (err) {
   435→      toast({ title: "خطأ", description: err instanceof Error ? err.message : "خطأ", variant: "destructive" });
   436→    } finally {
   437→      setCreatingGroup(false);
   438→    }
   439→  }

   461→      if (!res.ok) throw new Error((data as { error?: string }).error || "فشل التحديث");

   468→      toast({ title: "خطأ", description: err instanceof Error ? err.message : "خطأ", variant: "destructive" });
   469→    } finally {
