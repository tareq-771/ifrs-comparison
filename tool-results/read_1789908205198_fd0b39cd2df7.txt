    91→function makeSession() {
    92→  const jar = new Map<string, string>();
    93→  const cookieHeader = () => [...jar.entries()].map(([k, v]) => `${k}=${v}`).join("; ");
    94→  async function capture(res: Response): Promise<void> {
    95→    const setCookies = typeof res.headers.getSetCookie === "function" ? res.headers.getSetCookie() : [];
    96→    for (const sc of setCookies) {
    97→      const [pair] = sc.split(";");
    98→      const idx = pair.indexOf("=");
    99→      if (idx > 0) jar.set(pair.slice(0, idx).trim(), pair.slice(idx + 1).trim());
   100→    }
   101→  }
   102→  async function login(username: string, password: string): Promise<boolean> {
   103→    const csrfRes = await fetch(`${BASE}/api/auth/csrf`, { redirect: "manual" });
   104→    await capture(csrfRes);
   105→    const { csrfToken } = (await csrfRes.json()) as { csrfToken: string };
   106→    const body = new URLSearchParams({ csrfToken, username, password, json: "true" });
   107→    const res = await fetch(`${BASE}/api/auth/callback/credentials`, {
   108→      method: "POST",
   109→      headers: { "Content-Type": "application/x-www-form-urlencoded", Cookie: cookieHeader() },
   110→      body: body.toString(),
   111→      redirect: "manual",
   112→    });
   113→    await capture(res);
   114→    return !!(await whoami())?.id;
   115→  }
   116→  async function whoami(): Promise<{ id?: string; username?: string } | null> {
   117→    const res = await fetch(`${BASE}/api/auth/session`, { headers: { Cookie: cookieHeader() }, cache: "no-store" });
   118→    await capture(res);
   119→    if (!res.ok) return null;
   120→    const j = (await res.json()) as { user?: { id?: string; username?: string } };
   121→    return j?.user ?? null;
   122→  }
   123→  async function api<T = unknown>(p: string, init?: RequestInit): Promise<{ status: number; body: T }> {
   124→    const res = await fetch(`${BASE}${p}`, {
   125→      ...init,
   126→      headers: { "Content-Type": "application/json", Cookie: cookieHeader(), ...(init?.headers ?? {}) },
   127→    });
   128→    await capture(res);
   129→    const body = (await res.json().catch(() => ({}))) as T;
   130→    return { status: res.status, body };
   131→  }
   132→  const drop = () => jar.clear();
   133→  return { login, whoami, api, drop, cookieHeader };
   134→}

   149→async function startServer(): Promise<void> {
   150→  const proc = spawn("bunx", ["next", "dev", "-p", String(PORT)], {
   151→    cwd: ROOT,
   152→    env: { ...serverEnv(), NODE_ENV: process.env.NODE_ENV ?? "development" } as NodeJS.ProcessEnv,
   153→    stdio: "pipe" as const,
   154→  });
   155→  serverProc = proc;
   156→  proc.stdout?.on("data", (d) => {
   157→    serverLog += String(d);
   158→  });
   159→  proc.stderr?.on("data", (d) => {
   160→    serverLog += String(d);
   161→  });
   162→  const deadline = Date.now() + 150_000;
   163→  while (Date.now() < deadline) {
   164→    try {
   165→      // /api/system/status معفاة من حراس الصيانة — تعمل حتى لو أقلع الخادم
   166→      // في RECOVERY_REQUIRED بعد crash (وهذا متوقع في اختبارات 11/12)
   167→      const res = await fetch(`${BASE}/api/system/status`, { cache: "no-store" });
   168→      if (res.ok) return;
   169→    } catch {
   170→      /* not ready */
   171→    }
   172→    await sleep(800);
   173→  }
   174→  throw new Error("الخادم المعزول لم يجهز — السجل:\n" + serverLog.slice(-3000));
   175→}

   177→async function stopServer(): Promise<void> {
   178→  if (!serverProc) return;
   179→  const p = serverProc;
   180→  serverProc = null;
   181→  p.kill("SIGKILL");
   182→  // قتل السلسلة كاملة: bunx (parent) قد يموت ويرتب next-server (grandchild) —
   183→  // يبقى ممسكًا بالمنفذ ⇒ EADDRINUSE في الإقلاع التالي. نصطاد حامل المنفذ صراحة.
   184→  const killPort = spawnSync("bash", ["-c", `ss -ltnp 2>/dev/null | grep ':${PORT} ' | grep -oP 'pid=\\K[0-9]+' | xargs -r kill -9`], { encoding: "utf8" });
   185→  void killPort;
   186→  spawnSync("pkill", ["-f", `next dev -p ${PORT}`], { encoding: "utf8" });
   187→  await sleep(800);
   188→}

   271→async function createBackupAndDrill(s: Session): Promise<string> {

   277→    bk = await s.api<{ backupId: string; retryAfterMs?: number }>("/api/backups", { method: "POST" });

   282→    throw new Error(`فشل Drill: ${JSON.stringify(drill.body)}`);
   283→  }
