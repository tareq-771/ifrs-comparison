   126→const AR_DIGITS = "٠١٢٣٤٥٦٧٨٩";

   128→export function normalizeName(v: unknown): string {
   129→  let t = String(v ?? "").normalize("NFKC");
   130→  t = t.replace(/[\u200B-\u200F\uFEFF]/g, "");
   131→  t = t.replace(/[\u064B-\u065F\u0670\u0640]/g, "");
   132→  t = t.replace(/[^\w\u0600-\u06FF\s]/g, " ");
   133→  t = t.replace(/\s+/g, " ").trim().toLowerCase();
   134→  t = t.replace(/ة/g, "ه").replace(/ى/g, "ي").replace(/[أإآ]/g, "ا");
   135→  return t;
   136→}

   138→export function parseAmount(v: unknown): number {
   139→  if (v == null || v === "") return 0;
   140→  if (typeof v === "number") return v;
   141→  let s = String(v)
   142→    .replace(/[٠-٩]/g, (d) => String(AR_DIGITS.indexOf(d)))
   143→    .replace(/[,،\s]/g, "");
   144→  const neg = s[0] === "-";
   145→  s = s.replace(/[-*]+/g, "");
   146→  const x = parseFloat(s);
   147→  return Number.isNaN(x) ? 0 : neg ? -x : x;
   148→}

   150→export function parseAccountNum(v: unknown): string | null {
   151→  if (v == null || v === "") return null;
   152→  if (typeof v === "number") return String(Math.round(v));
   153→  const s = String(v)
   154→    .replace(/[٠-٩]/g, (d) => String(AR_DIGITS.indexOf(d)))
   155→    .trim();
   156→  const d = s.replace(/\D/g, "");
   157→  return d ? d.replace(/^0+(?=.)/, "") || "0" : null;
   158→}

   164→export function readExcelFile(file: File): Promise<FileData> {
   165→  return new Promise((resolve, reject) => {
   166→    const fr = new FileReader();
   167→    fr.onload = (e) => {
   168→      try {
   169→        const buf = e.target?.result;
   170→        if (!buf) throw new Error("ملف فارغ");
   171→        const wb = XLSX.read(buf, { type: "array" });
   172→        const sh = wb.SheetNames[0];
   173→        const rows = XLSX.utils.sheet_to_json<unknown[]>(wb.Sheets[sh], {
   174→          header: 1, raw: true, defval: null,
   175→        });
   176→        let hi = -1;
   177→        const cap = Math.min(rows.length, 30);
   178→        for (let i = 0; i < cap; i++) {
   179→          const cells = (rows[i] || []).map((c) => String(c ?? "").trim());
   180→          for (let j = 0; j < cells.length; j++) {
   181→            if (cells[j].indexOf("اسم الحساب") >= 0) { hi = i; break; }
   182→          }
   183→          if (hi >= 0) break;
   184→        }
   185→        if (hi < 0) throw new Error("لا يوجد عمود «اسم الحساب»");
   186→        const hdr = rows[hi].map((c) => String(c ?? "").trim());
   187→        const sub = (rows[hi + 1] || []).map((c) => String(c ?? "").trim());
   188→        let cN = 1, cD = -1, cNum = -1;
   189→        for (let j = 0; j < hdr.length; j++) {
   190→          const h = hdr[j];
   191→          if (h.indexOf("اسم الحساب") >= 0) cN = j;
   192→          if (sub[j] === "مدين" || h === "مدين") cD = j;
   193→        }
   194→        // Detect account number column AFTER name column — avoid picking "اسم الحساب"
   195→        for (let j = 0; j < hdr.length; j++) {
   196→          const h = hdr[j];
   197→          if (j === cN) continue; // skip the name column
   198→          if (
   199→            h.indexOf("رقم الحساب") >= 0 ||
   200→            h.indexOf("رقم") >= 0 ||
   201→            h === "الحساب" ||
   202→            h === "الرقم" ||
   203→            h === "حساب"
   204→          ) { cNum = j; break; }
   205→        }
   206→        if (cNum < 0) cNum = 0; // fallback: column 0
   207→        if (cD < 0) {
   208→          for (let j = 0; j < hdr.length; j++) {
   209→            if (hdr[j] === "المبلغ") { cD = j; break; }
   210→          }
   211→        }
   212→        if (cD < 0) cD = cN + 2;
   213→        const A: AccountRow[] = [];
   214→        for (let i = hi + 1; i < rows.length; i++) {
   215→          const r = rows[i] || [];
   216→          const g = (j: number) => (j >= 0 && j < r.length ? r[j] : null);
   217→          const n = String(g(cN) ?? "").trim();
   218→          if (!n || n === "*") continue;
   219→          if (n.indexOf("الإجمالي") >= 0 || n.indexOf("الرصيد") >= 0) continue;
   220→          A.push({ nm: n, nk: normalizeName(n), num: parseAccountNum(g(cNum)), m: parseAmount(g(cD)), d: parseAmount(g(cD + 1)) });
   221→        }
   222→        resolve({ A, headers: hdr, nameCol: cN, numCol: cNum, debitCol: cD, headerRow: hi, rawRows: rows });
   223→      } catch (err) {
   224→        reject(err instanceof Error ? err : new Error(String(err)));
   225→      }
   226→    };
   227→    fr.onerror = () => reject(new Error("فشل قراءة الملف"));
   228→    fr.readAsArrayBuffer(file);
   229→  });
   230→}

   232→export function reExtract(file: FileData, nameCol: number, debitCol: number): AccountRow[] {
   233→  // Files restored from a saved report (فتح تقرير محفوظ) carry pre-extracted rows
   234→  // and NO raw sheet (rawRows = []). Re-parsing would yield an empty array and wipe
   235→  // out the base-account selection — keep the stored rows as-is in that case.
   236→  if (!file.rawRows || file.rawRows.length === 0) return file.A;
   237→  const rows = file.rawRows;
   238→  const hi = file.headerRow;
   239→  const numCol = file.numCol ?? 0;
   240→  const A: AccountRow[] = [];
   241→  for (let i = hi + 1; i < rows.length; i++) {
   242→    const r = rows[i] || [];
   243→    const g = (j: number) => (j >= 0 && j < r.length ? r[j] : null);
   244→    const n = String(g(nameCol) ?? "").trim();
   245→    if (!n || n === "*") continue;
   246→    if (n.indexOf("الإجمالي") >= 0 || n.indexOf("الرصيد") >= 0) continue;
   247→    A.push({ nm: n, nk: normalizeName(n), num: parseAccountNum(g(numCol)), m: parseAmount(g(debitCol)), d: parseAmount(g(debitCol + 1)) });
   248→  }
   249→  return A;
   250→}

   256→export function matchFiles(f1: FileData, f2: FileData): MatchedRow[] {
   257→  const out: MatchedRow[] = [];
   258→  const m2: Record<string, AccountRow[]> = {};
   259→  for (const a of f2.A) { (m2[a.nk] ||= []).push(a); }
   260→  for (const a1 of f1.A) {
   261→    const q = m2[a1.nk];
   262→    out.push({ nk: a1.nk, a1, a2: q && q.length ? q.shift()! : null, st: "مطابق" });
   263→  }
   264→  for (const k in m2) for (const a2 of m2[k]) out.push({ nk: k, a1: null, a2, st: "مطابق" });
   265→  for (const r of out) {
   266→    r.st = r.a1 && r.a2 ? "مطابق" : r.a1 ? "في الملف الأول فقط" : "في الملف الثاني فقط";
   267→  }
   268→  return out;
   269→}

   271→/* ════════════════════════════════════════════════════════════════════════════
   272→ * Chart-of-Accounts Matching — مطابقة دليل الحسابات
   273→ *
   274→ * Matches accounts by NAME AND NUMBER (OR logic):
   275→ *  - Name match   — normalized name similarity >= threshold (default 90%)
   276→ *  - Number match — account numbers are identical (exact match)
   277→ *
   278→ * Priority: both > number-only > name-only.
   279→ * Statuses: مطابق / تغير اسم / تغير رقم / تغير الموقع / في الملف الأول فقط / في الملف الثاني فقط
   280→ * ════════════════════════════════════════════════════════════════════════════ */

   282→/** Levenshtein edit distance between two strings (classic DP). */
   283→function levenshtein(a: string, b: string): number {
   284→  if (a === b) return 0;
   285→  if (!a.length) return b.length;
   286→  if (!b.length) return a.length;
   287→  let prev = new Array<number>(b.length + 1);
   288→  let curr = new Array<number>(b.length + 1);
   289→  for (let j = 0; j <= b.length; j++) prev[j] = j;
   290→  for (let i = 1; i <= a.length; i++) {
   291→    curr[0] = i;
   292→    for (let j = 1; j <= b.length; j++) {
   293→      const cost = a.charCodeAt(i - 1) === b.charCodeAt(j - 1) ? 0 : 1;
   294→      curr[j] = Math.min(curr[j - 1] + 1, prev[j] + 1, prev[j - 1] + cost);
   295→    }
   296→    const tmp = prev; prev = curr; curr = tmp;
   297→  }
   298→  return prev[b.length];
   299→}
   300→
   301→/** Name similarity ratio in [0, 1]. Levenshtein normalized by max length. */
   302→export function nameSimilarity(a: string, b: string): number {
   303→  if (!a && !b) return 0;
   304→  if (!a || !b) return 0;
   305→  if (a === b) return 1;
   306→  const maxLen = Math.max(a.length, b.length);
   307→  if (maxLen === 0) return 0;
   308→  return 1 - levenshtein(a, b) / maxLen;
   309→}

   311→/** Character-level LCS-based diff. Returns segments of {text, op}. */
   312→export type DiffOp = "same" | "add" | "del";
   313→export interface DiffSegment { text: string; op: DiffOp; }
   314→
   315→export function diffStrings(a: string, b: string): DiffSegment[] {
   316→  if (!a && !b) return [];
   317→  if (!a) return [{ text: b, op: "add" }];
   318→  if (!b) return [{ text: a, op: "del" }];
   319→  if (a === b) return [{ text: a, op: "same" }];
   320→  const m = a.length, n = b.length;
   321→  const dp: number[][] = Array.from({ length: m + 1 }, () => new Array<number>(n + 1).fill(0));
   322→  for (let i = 1; i <= m; i++) {
   323→    for (let j = 1; j <= n; j++) {
   324→      if (a.charCodeAt(i - 1) === b.charCodeAt(j - 1)) dp[i][j] = dp[i - 1][j - 1] + 1;
   325→      else dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
   326→    }
   327→  }
   328→  const segs: DiffSegment[] = [];
   329→  let i = m, j = n;
   330→  while (i > 0 || j > 0) {
   331→    if (i > 0 && j > 0 && a.charCodeAt(i - 1) === b.charCodeAt(j - 1)) {
   332→      segs.push({ text: a[i - 1], op: "same" }); i--; j--;
   333→    } else if (j > 0 && (i === 0 || dp[i][j - 1] >= dp[i - 1][j])) {
   334→      segs.push({ text: b[j - 1], op: "add" }); j--;
   335→    } else {
   336→      segs.push({ text: a[i - 1], op: "del" }); i--;
   337→    }
   338→  }
   339→  segs.reverse();
   340→  const merged: DiffSegment[] = [];
   341→  for (const s of segs) {
   342→    const last = merged[merged.length - 1];
   343→    if (last && last.op === s.op) last.text += s.text;
   344→    else merged.push({ ...s });
   345→  }
   346→  return merged;
   347→}

   349→/** Compact human-readable diff summary for Excel: "−X +Y" or "مطابق تام". */
   350→export function summarizeDiff(a: string, b: string): string {
   351→  if (!a || !b) return "—";
   352→  if (a === b) return "مطابق تام";
   353→  const diff = diffStrings(a, b);
   354→  let removed = "";
   355→  let added = "";
   356→  for (const s of diff) {
   357→    if (s.op === "del") removed += s.text;
   358→    else if (s.op === "add") added += s.text;
   359→  }
   360→  const parts: string[] = [];
   361→  if (removed) parts.push(`−${removed}`);
   362→  if (added) parts.push(`+${added}`);
   363→  return parts.length ? parts.join(" ") : "مطابق تام";
   364→}

   366→/**
   367→ * Account category by leading digit of account number.
   368→ *
   369→ * 4-category classification (per user's exact specification):
   370→ *   1 → أصول       (assets)
   371→ *   2 → خصوم       (liabilities)
   372→ *   3 → مصروفات    (expenses — per user request, NOT equity)
   373→ *   4 → إيرادات    (revenues)
   374→ *   5 → مصروفات    (expenses — folded under expenses)
   375→ *   6 → مصروفات    (taxes folded under expenses)
   376→ *   7 → إيرادات    (other revenues folded under revenues)
   377→ *   other → غير محدد

   379→export function accountCategory(num: string | null): string {
   380→  if (!num) return "غير محدد";
   381→  const d = num.charAt(0);
   382→  switch (d) {
   383→    case "1": return "أصول";
   384→    case "2": return "خصوم";
   385→    case "3":
   386→    case "5":
   387→    case "6": return "مصروفات";
   388→    case "4":
   389→    case "7": return "إيرادات";
   390→    default: return "غير محدد";
   391→  }
   392→}

   394→/**
   395→ * Match two charts of accounts by NAME AND NUMBER (OR logic).
   396→ * A pair is a candidate if EITHER number matches exactly OR name sim >= threshold.
   397→ * Priority: both > number-only > name-only.
   398→ */
   399→export function matchChartOfAccounts(
   400→  f1: FileData,
   401→  f2: FileData,
   402→  threshold = 0.9
   403→): MatchedRow[] {
   404→  const a1s = f1.A.slice();
   405→  const a2s = f2.A.slice();
   406→  type Pair = { i: number; j: number; sim: number; numMatch: boolean; nameMatch: boolean };
   407→  const pairs: Pair[] = [];
   408→  for (let i = 0; i < a1s.length; i++) {
   409→    const a1 = a1s[i];
   410→    if (!a1.nk && !a1.num) continue;
   411→    for (let j = 0; j < a2s.length; j++) {
   412→      const a2 = a2s[j];
   413→      if (!a2.nk && !a2.num) continue;
   414→      const numMatch = !!a1.num && !!a2.num && a1.num === a2.num;
   415→      const sim = nameSimilarity(a1.nk, a2.nk);
   416→      const nameMatch = sim >= threshold;
   417→      if (numMatch || nameMatch) {
   418→        pairs.push({ i, j, sim, numMatch, nameMatch });
   419→      }
   420→    }
   421→  }
   422→  function pairPriority(p: Pair): number {
   423→    if (p.numMatch && p.nameMatch) return 3;
   424→    if (p.numMatch) return 2;
   425→    return 1;
   426→  }
   427→  pairs.sort((x, y) => {
   428→    const px = pairPriority(x);
   429→    const py = pairPriority(y);
   430→    if (px !== py) return py - px;
   431→    return y.sim - x.sim;
   432→  });
   433→
   434→  const used1 = new Set<number>();
   435→  const used2 = new Set<number>();
   436→  const out: MatchedRow[] = [];
   437→
   438→  for (const p of pairs) {
   439→    if (used1.has(p.i) || used2.has(p.j)) continue;
   440→    used1.add(p.i);
   441→    used2.add(p.j);
   442→    const a1 = a1s[p.i];
   443→    const a2 = a2s[p.j];
   444→    const cat1 = accountCategory(a1.num);
   445→    const cat2 = accountCategory(a2.num);
   446→    const numChanged = (a1.num ?? null) !== (a2.num ?? null);
   447→    const catChanged = cat1 !== cat2;
   448→    const nameChanged = a1.nk !== a2.nk;
   449→    let matchBy: "both" | "name" | "number";
   450→    if (p.numMatch && p.nameMatch) matchBy = "both";
   451→    else if (p.numMatch) matchBy = "number";
   452→    else matchBy = "name";
   453→
   454→    // ── 4-sheet classification (per user's new requirement) ──
   455→    // Sheet 1 «مطابق الاسم والرقم»:   name AND number both match (priority to name)
   456→    // Sheet 2 «مطابق الاسم مختلف الرقم»: name matches, number differs (any category)
   457→    // Sheet 3 «غير مطابق الاسم»:        unmatched (in one file only) — handled below
   458→    // Sheet 4 «التغير الجوهري»:         category CHANGED between the two files
   459→    //                                    (e.g. account moved from أصول to خصوم,
   460→    //                                     or from مصروفات to إيرادات)
   461→    let st: MatchStatus;
   462→    if (catChanged) {
   463→      // Category change is the fundamental change → Sheet 4
   464→      st = "تغير جوهري";
   465→    } else if (!numChanged && !nameChanged) {
   466→      // Both name AND number match → Sheet 1
   467→      st = "مطابق";
   468→    } else if (!nameChanged && numChanged) {
   469→      // Name matches, number differs, same category → Sheet 2
   470→      st = "مطابق الاسم مختلف الرقم";
   471→    } else {
   472→      // nameChanged && numChanged (both differ) — matched by one criterion only
   473→      // Same category, but both name AND number differ
   474→      // → keep in Sheet 2 if matched by name (name is the priority criterion)
   475→      // → otherwise keep in Sheet 2 too (the user wants name-priority matching)
   476→      st = "مطابق الاسم مختلف الرقم";
   477→    }
   478→    // Compute amount change + change percentage (debit−credit balance)
   479→    const amt1 = a1.m - a1.d;
   480→    const amt2 = a2.m - a2.d;
   481→    const amtChange = amt2 - amt1;
   482→    const amtChangePct = amt1 !== 0 ? (amtChange / Math.abs(amt1)) * 100 : null;
   483→    out.push({
   484→      nk: a1.nk, a1, a2, st,
   485→      sim: Math.round(p.sim * 100),
   486→      cat1, cat2,
   487→      numChanged, catChanged, nameChanged, matchBy,
   488→      amt1, amt2, amtChange, amtChangePct,
   489→    });
   490→  }
   491→
   492→  for (let i = 0; i < a1s.length; i++) {
   493→    if (used1.has(i)) continue;
   494→    const a1 = a1s[i];
   495→    out.push({
   496→      nk: a1.nk, a1, a2: null, st: "في الملف الأول فقط",
   497→      sim: null, cat1: accountCategory(a1.num), cat2: undefined,
   498→      numChanged: false, catChanged: false, nameChanged: false, matchBy: undefined,
   499→      amt1: a1.m - a1.d, amt2: null, amtChange: null, amtChangePct: null,
   500→    });
   501→  }
   502→  for (let j = 0; j < a2s.length; j++) {
   503→    if (used2.has(j)) continue;
   504→    const a2 = a2s[j];
   505→    out.push({
   506→      nk: a2.nk, a1: null, a2, st: "في الملف الثاني فقط",
   507→      sim: null, cat1: undefined, cat2: accountCategory(a2.num),
   508→      numChanged: false, catChanged: false, nameChanged: false, matchBy: undefined,
   509→      amt1: null, amt2: a2.m - a2.d, amtChange: null, amtChangePct: null,
   510→    });
   511→  }
   512→
   513→  // Sort: matched → name-match-num-diff → fundamental → unmatched
   514→  const order: Record<MatchStatus, number> = {
   515→    "مطابق": 0,
   516→    "مطابق الاسم مختلف الرقم": 1,
   517→    "تغير جوهري": 2,
   518→    "في الملف الأول فقط": 3,
   519→    "في الملف الثاني فقط": 4,
   520→    "غير مطابق": 5,
   521→  };
   522→  out.sort((x, y) => {
   523→    const o = order[x.st] - order[y.st];
   524→    if (o !== 0) return o;
   525→    const sx = x.sim ?? -1;
   526→    const sy = y.sim ?? -1;
   527→    return sy - sx;
   528→  });
   529→  return out;
   530→}

   548→      opts.push({ nk: a.nk, nm: a.nm, num: a.num ?? null, sc });

   555→/* ──────────────────────────────────────────────────────────────────────────
   556→ * Categorization — prefix-based (IAS 1 function-of-expense)
   557→ * ────────────────────────────────────────────────────────────────────────── */
   558→
   559→export function categorize(out: MatchedRow[], settings: CategorySettings): Categorized {
   560→  const { base, costPrefix, sellPrefix, adminPrefix, otherOpPrefix, financePrefix, taxPrefix, ociPrefix } = settings;
   561→
   562→  const ba = base ? out.find((r) => r.nk === base) ?? null : null;
   563→  const netRow = findNetRow(out);
   564→  const baseNum = (ba?.a1?.num ?? ba?.a2?.num ?? null) ?? null;
   565→
   566→  const cat: Categorized = {
   567→    sales: [], cost: [], sell: [], adm: [], oth: [], fin: [], tax: [], oci: [], othRev: [], net: [],
   568→  };
   569→
   570→  for (const r of out) {
   571→    // Exclude base and net-income summary rows from categorization
   572→    if (r === ba || r === netRow) continue;
   573→
   574→    const num = String((r.a1?.num ?? r.a2?.num ?? "") as string);
   575→
   576→    // Revenue tree — accounts whose number starts with base number
   577→    if (baseNum && num && num.indexOf(baseNum) === 0) { cat.sales.push(r); continue; }
   578→    // Cost of sales
   579→    if (costPrefix && num && num.indexOf(costPrefix) === 0) { cat.cost.push(r); continue; }
   580→    // Selling & distribution
   581→    if (sellPrefix && num && num.indexOf(sellPrefix) === 0) { cat.sell.push(r); continue; }
   582→    // General & administrative
   583→    if (adminPrefix && num && num.indexOf(adminPrefix) === 0) { cat.adm.push(r); continue; }
   584→    // Other operating
   585→    if (otherOpPrefix && num && num.indexOf(otherOpPrefix) === 0) { cat.oth.push(r); continue; }
   586→    // Finance
   587→    if (financePrefix && num && num.indexOf(financePrefix) === 0) { cat.fin.push(r); continue; }
   588→    // Tax
   589→    if (taxPrefix && num && num.indexOf(taxPrefix) === 0) { cat.tax.push(r); continue; }
   590→    // OCI
   591→    if (ociPrefix && num && num.indexOf(ociPrefix) === 0) { cat.oci.push(r); continue; }
   592→    // Unprefixed accounts starting with "3" → other operating
   593→    if (num && num.indexOf("3") === 0) { cat.oth.push(r); continue; }
   594→    // Everything else → other income
   595→    cat.othRev.push(r);
   596→  }
   597→
   598→  const byNum = (a: MatchedRow, b: MatchedRow) => {
   599→    const x = String((a.a1?.num ?? a.a2?.num ?? "") as string);
   600→    const y = String((b.a1?.num ?? b.a2?.num ?? "") as string);
   601→    return x < y ? -1 : x > y ? 1 : 0;
   602→  };
   603→  cat.sales.sort(byNum); cat.cost.sort(byNum); cat.sell.sort(byNum);
   604→  cat.adm.sort(byNum); cat.oth.sort(byNum); cat.fin.sort(byNum);
   605→  cat.tax.sort(byNum); cat.oci.sort(byNum); cat.othRev.sort(byNum);
   606→  return cat;
   607→}

   609→/** Find the "صافي الدخل" summary row (for reconciliation). */
   610→export function findNetRow(out: MatchedRow[]): MatchedRow | null {
   611→  for (const r of out) {
   612→    const n = normalizeName((r.a1 ?? r.a2)?.nm ?? "");
   613→    if (n.indexOf("صافي الدخل") >= 0 || n.indexOf("صافي دخ") >= 0) return r;
   614→  }
   615→  return null;
   616→}

   618→/* ──────────────────────────────────────────────────────────────────────────
   619→ * Leaf detection — parents are shown bold but excluded from totals
   620→ * ────────────────────────────────────────────────────────────────────────── */
   621→
   622→export interface RowLeafInfo {
   623→  leaf1: boolean;
   624→  leaf2: boolean;
   625→}
   626→
   627→export function leafInfo(rows: MatchedRow[]): RowLeafInfo[] {
   628→  return rows.map((r, i) => {
   629→    let leaf1 = false, leaf2 = false;
   630→    const a1 = r.a1;
   631→    if (a1 && a1.num) {
   632→      leaf1 = true;
   633→      for (let j = 0; j < rows.length; j++) {
   634→        if (j === i) continue;
   635→        const o = rows[j].a1;
   636→        if (!o || !o.num) continue;
   637→        if (o.num.length > a1.num.length && o.num.indexOf(a1.num) === 0) { leaf1 = false; break; }
   638→      }
   639→    }
   640→    const a2 = r.a2;
   641→    if (a2 && a2.num) {
   642→      leaf2 = true;
   643→      for (let j = 0; j < rows.length; j++) {
   644→        if (j === i) continue;
   645→        const o = rows[j].a2;
   646→        if (!o || !o.num) continue;
   647→        if (o.num.length > a2.num.length && o.num.indexOf(a2.num) === 0) { leaf2 = false; break; }
   648→      }
   649→    }
   650→    return { leaf1, leaf2 };
   651→  });
   652→}

   654→/* ──────────────────────────────────────────────────────────────────────────
   655→ * Totals — IFRS cascade with credit/debit sign convention
   656→ * ────────────────────────────────────────────────────────────────────────── */
   657→
   658→const r2 = (n: number) => Math.round(n * 100) / 100;
   659→
   660→/**
   661→ * Sum LEAF accounts of a category on one file side.
   662→ * @param isRevenue  true → credit-positive (دائن − مدين), false → debit-positive (مدين − دائن)
   663→ */
   664→function sumLeaves(
   665→  rows: MatchedRow[], info: RowLeafInfo[], side: "a1" | "a2", isRevenue: boolean
   666→): number {
   667→  let s = 0;
   668→  for (let i = 0; i < rows.length; i++) {
   669→    const isLeaf = side === "a1" ? info[i].leaf1 : info[i].leaf2;
   670→    if (!isLeaf) continue;
   671→    const a = rows[i][side]!;
   672→    s += isRevenue ? a.d - a.m : a.m - a.d;
   673→  }
   674→  return r2(s);
   675→}

   677→/**
   678→ * Compute all income-statement totals.
   679→ * @param baseRow  The selected net-sales account (its OWN balance = revenue).
   680→ */
   681→export function calcTotals(
   682→  cat: Categorized,
   683→  baseRow: MatchedRow | null
   684→): Totals {
   685→  // Revenue = base account's own balance (credit − debit)
   686→  const rev1 = baseRow?.a1 ? r2(baseRow.a1.d - baseRow.a1.m) : 0;
   687→  const rev2 = baseRow?.a2 ? r2(baseRow.a2.d - baseRow.a2.m) : 0;
   688→
   689→  const iCost = leafInfo(cat.cost);
   690→  const iSell = leafInfo(cat.sell);
   691→  const iAdm = leafInfo(cat.adm);
   692→  const iOth = leafInfo(cat.oth);
   693→  const iFin = leafInfo(cat.fin);
   694→  const iTax = leafInfo(cat.tax);
   695→  const iOci = leafInfo(cat.oci);
   696→  const iOthRev = leafInfo(cat.othRev);
   697→
   698→  const cost1 = sumLeaves(cat.cost, iCost, "a1", false);
   699→  const cost2 = sumLeaves(cat.cost, iCost, "a2", false);
   700→  const sell1 = sumLeaves(cat.sell, iSell, "a1", false);
   701→  const sell2 = sumLeaves(cat.sell, iSell, "a2", false);
   702→  const adm1 = sumLeaves(cat.adm, iAdm, "a1", false);
   703→  const adm2 = sumLeaves(cat.adm, iAdm, "a2", false);
   704→  const oth1 = sumLeaves(cat.oth, iOth, "a1", false);
   705→  const oth2 = sumLeaves(cat.oth, iOth, "a2", false);
   706→  const othRev1 = sumLeaves(cat.othRev, iOthRev, "a1", true);
   707→  const othRev2 = sumLeaves(cat.othRev, iOthRev, "a2", true);
   708→  const fin1 = sumLeaves(cat.fin, iFin, "a1", false);
   709→  const fin2 = sumLeaves(cat.fin, iFin, "a2", false);
   710→  const tax1 = sumLeaves(cat.tax, iTax, "a1", false);
   711→  const tax2 = sumLeaves(cat.tax, iTax, "a2", false);
   712→  const oci1 = sumLeaves(cat.oci, iOci, "a1", true);
   713→  const oci2 = sumLeaves(cat.oci, iOci, "a2", true);
   714→
   715→  const gross1 = r2(rev1 - cost1);
   716→  const gross2 = r2(rev2 - cost2);
   717→  const opex1 = r2(sell1 + adm1 + oth1);
   718→  const opex2 = r2(sell2 + adm2 + oth2);
   719→  const oper1 = r2(gross1 - opex1);
   720→  const oper2 = r2(gross2 - opex2);
   721→  const pbt1 = r2(oper1 + othRev1 - fin1);
   722→  const pbt2 = r2(oper2 + othRev2 - fin2);
   723→  const net1 = r2(pbt1 - tax1);
   724→  const net2 = r2(pbt2 - tax2);
   725→  const comp1 = r2(net1 + oci1);
   726→  const comp2 = r2(net2 + oci2);
   727→
   728→  return {
   729→    rev1, rev2, cost1, cost2, gross1, gross2,
   730→    sell1, sell2, adm1, adm2, oth1, oth2, opex1, opex2,
   731→    oper1, oper2, othRev1, othRev2, fin1, fin2,
   732→    pbt1, pbt2, tax1, tax2, net1, net2, oci1, oci2, comp1, comp2,
   733→  };
   734→}

   763→/* ──────────────────────────────────────────────────────────────────────────
   764→ * Change analysis — desirability & percentage
   765→ * ──────────────────────────────────────────────────────────────────────────
   766→ *
   767→ * For EXPENSE accounts (prefix 3, debit-positive):
   768→ *   - Decrease (chg < 0) = savings = مرغوب  (desirable)
   769→ *   - Increase (chg > 0) = overspend = غير مرغوب (undesirable)
   770→ *
   771→ * For REVENUE accounts (prefix 4, credit-positive):
   772→ *   - Increase (chg > 0) = more income = مرغوب  (desirable)
   773→ *   - Decrease (chg < 0) = lost income = غير مرغوب (undesirable)
   774→ *
   775→ * Change percentage = chg / base period (v1 = comparative/older period)
   776→ */
   777→
   778→export type ChangeStatus =
   779→  | "مرغوب نمو"
   780→  | "غير مرغوب انخفاض"
   781→  | "مرغوب وفرة"
   782→  | "غير مرغوب زيادة"
   783→  | "—";
   784→
   785→export function computeChangeStatus(
   786→  chg: number | null,
   787→  isRevenue: boolean
   788→): ChangeStatus {
   789→  if (chg == null || chg === 0) return "—";
   790→  if (isRevenue) {
   791→    // Revenue: increase = good (growth), decrease = bad (decline)
   792→    return chg > 0 ? "مرغوب نمو" : "غير مرغوب انخفاض";
   793→  }
   794→  // Expense: decrease = good (savings), increase = bad (overspend)
   795→  return chg < 0 ? "مرغوب وفرة" : "غير مرغوب زيادة";
   796→}
   797→
   798→export function computeChangePct(
   799→  chg: number | null,
   800→  v1: number | null
   801→): number | null {
   802→  if (chg == null || v1 == null || v1 === 0) return null;
   803→  return chg / v1;
   804→}
   805→
   806→/* ──────────────────────────────────────────────────────────────────────────
   807→ * Comparison modes
   808→ * ──────────────────────────────────────────────────────────────────────────
   809→ *
   810→ * Mode 1 — "period": Current period vs Previous period (change analysis)
   811→ * Mode 2 — "monthCumulative": Current month vs Year-to-date cumulative
   812→ *          (month's contribution to the annual total)
   813→ *
   814→ * In mode 2, the key metric is monthRatio = month_value / cumulative_value.
   815→ * A "normal" month should represent ~1/numMonths of the annual total.
   816→ * Desirability:
   817→ *   Revenue: month ratio > expected = مرغوب (strong month)
   818→ *   Expense: month ratio > expected = غير مرغوب (high expense month)
   819→ */
   820→
   821→export type CompareMode = "period" | "monthCumulative" | "chart";
   822→
   823→/** Month/cumulative ratio = month value (v2) / cumulative value (v1). */
   824→export function computeMonthRatio(
   825→  v2: number | null,
   826→  v1: number | null
   827→): number | null {
   828→  if (v2 == null || v1 == null || v1 === 0) return null;
   829→  return v2 / v1;
   830→}
   831→
   832→/** Remaining = cumulative − month (what other months contributed). */
   833→export function computeRemaining(
   834→  v1: number | null,
   835→  v2: number | null
   836→): number | null {
   837→  if (v1 == null || v2 == null) return null;
   838→  return r2(v1 - v2);
   839→}
   840→
   841→/** Average cumulative = cumulative value / number of months. */
   842→export function computeAvgCumulative(
   843→  v1: number | null,
   844→  numMonths: number
   845→): number | null {
   846→  if (v1 == null || numMonths === 0) return null;
   847→  return r2(v1 / numMonths);
   848→}
   849→
   850→/**
   851→ * Ratio of month value to the cumulative average.
   852→ * month / (cumulative / numMonths) = month × numMonths / cumulative.
   853→ * Values > 1 mean the month exceeds the average.
   854→ */
   855→export function computeMonthToAvgRatio(
   856→  v2: number | null,
   857→  avgCumulative: number | null
   858→): number | null {
   859→  if (v2 == null || avgCumulative == null || avgCumulative === 0) return null;
   860→  return v2 / avgCumulative;
   861→}
   862→
   863→/** Absolute difference: month value − average cumulative. */
   864→export function computeMonthAvgDiff(
   865→  v2: number | null,
   866→  avgCumulative: number | null
   867→): number | null {
   868→  if (v2 == null || avgCumulative == null) return null;
   869→  return r2(v2 - avgCumulative);
   870→}
   871→
   872→/** Percentage difference from average: (month − avg) / avg. */
   873→export function computeMonthAvgDiffPct(
   874→  v2: number | null,
   875→  avgCumulative: number | null
   876→): number | null {
   877→  if (v2 == null || avgCumulative == null || avgCumulative === 0) return null;
   878→  return (v2 - avgCumulative) / avgCumulative;
   879→}
   880→
   881→/**
   882→ * Desirability of the month's concentration vs the expected average.
   883→ * @param monthRatio  month/cumulative ratio (0..1)
   884→ * @param isRevenue   true for revenue, false for expense
   885→ * @param numMonths   number of months in the cumulative period (default 12)
   886→ */
   887→export function computeMonthStatus(
   888→  monthRatio: number | null,
   889→  isRevenue: boolean,
   890→  numMonths = 12
   891→): ChangeStatus {
   892→  if (monthRatio == null || monthRatio === 0) return "—";
   893→  const expected = 1 / numMonths;
   894→  if (Math.abs(monthRatio - expected) < 0.005) return "—";
   895→  if (isRevenue) {
   896→    return monthRatio > expected ? "مرغوب نمو" : "غير مرغوب انخفاض";
   897→  }
   898→  return monthRatio > expected ? "غير مرغوب زيادة" : "مرغوب وفرة";
   899→}

  1437→const FIXED_ASSETS_KEYWORDS = ["اصول ثابته", "أصول ثابتة", "اثاث", "أثاث", "معدات", "سيارات", "مباني", "مبانى", "اراضي", "أراضي", "اجهزه", "أجهزة"];
  1438→const PAYABLES_KEYWORDS = ["دائنون", "ذمم دائن", "موردون", "موردين"];
  1439→const SHORT_TERM_DEBT_KEYWORDS = ["اوراق دفع", "أوراق دفع", "سحب على المكشوف", "تسهيلات بنكيه", "قروض قصيره", "قروض قصيرة"];
  1440→const LONG_TERM_DEBT_KEYWORDS = ["قروض طويله", "قروض طويلة", "سندات", "قروض حسنه"];

  1442→function hasKeyword(n: string, keywords: string[]): boolean {
  1443→  return keywords.some((k) => n.indexOf(k) >= 0);
  1444→}

  1446→/** Check if an account number matches a prefix (empty prefix = no match). */
  1447→function matchesPrefix(num: string, prefix: string): boolean {
  1448→  return !!prefix && num.indexOf(prefix) === 0;

  1451→/**
  1452→ * Parse a balance-sheet file and compute category totals.
  1453→ * Uses the same parsing as income-statement files (readExcelFile).
  1454→ * The `totals` are computed from LEAF accounts only (same logic as income statement).
  1455→ *
  1456→ * Detailed items (inventory, cash, etc.) are detected by PREFIX first;
  1457→ * if the prefix is empty, falls back to keyword-based name detection.
  1458→ */
  1459→export function categorizeBalanceSheet(
  1460→  file: FileData,
  1461→  settings: BalanceSheetSettings
  1462→): BalanceSheetTotals {
  1463→  const {
  1464→    currentAssetsPrefix, nonCurrentAssetsPrefix, currentLiabPrefix, nonCurrentLiabPrefix, equityPrefix,
  1465→    inventoryPrefix, cashPrefix, receivablesPrefix, fixedAssetsPrefix,
  1466→    payablesPrefix, shortTermDebtPrefix, longTermDebtPrefix,
  1467→  } = settings;
  1468→
  1469→  // Mark leaves
  1470→  const nums = file.A.map((a) => a.num).filter(Boolean) as string[];
  1471→  for (const a of file.A) {
  1472→    if (!a.num) { (a as any).leaf = true; continue; }
  1473→    (a as any).leaf = true;
  1474→    for (const n of nums) {
  1475→      if (n !== a.num && n.length > a.num.length && n.indexOf(a.num) === 0) {
  1476→        (a as any).leaf = false;
  1477→        break;
  1478→      }
  1479→    }
  1480→  }
  1481→
  1482→  let currentAssets = 0, nonCurrentAssets = 0;
  1483→  let currentLiabilities = 0, nonCurrentLiabilities = 0;
  1484→  let equity = 0;
  1485→  let inventory = 0, cash = 0, receivables = 0;
  1486→  let fixedAssets = 0, payables = 0, shortTermDebt = 0, longTermDebt = 0;
  1487→
  1488→  for (const a of file.A) {
  1489→    if (!(a as any).leaf) continue;
  1490→    const num = a.num ?? "";
  1491→    const n = normalizeName(a.nm);
  1492→    const debitBal = a.m - a.d;   // asset = debit − credit
  1493→    const creditBal = a.d - a.m;  // liability/equity = credit − debit
  1494→
  1495→    // ── Detailed item detection: prefix first, then keyword fallback ──
  1496→    // Inventory
  1497→    if (matchesPrefix(num, inventoryPrefix) || (!inventoryPrefix && hasKeyword(n, INVENTORY_KEYWORDS))) {
  1498→      inventory += debitBal;
  1499→    }
  1500→    // Cash & equivalents
  1501→    if (matchesPrefix(num, cashPrefix) || (!cashPrefix && hasKeyword(n, CASH_KEYWORDS))) {
  1502→      cash += debitBal;
  1503→    }
  1504→    // Receivables
  1505→    if (matchesPrefix(num, receivablesPrefix) || (!receivablesPrefix && hasKeyword(n, RECEIVABLES_KEYWORDS))) {
  1506→      receivables += debitBal;
  1507→    }
  1508→    // Fixed assets
  1509→    if (matchesPrefix(num, fixedAssetsPrefix) || (!fixedAssetsPrefix && hasKeyword(n, FIXED_ASSETS_KEYWORDS))) {
  1510→      fixedAssets += debitBal;
  1511→    }
  1512→    // Payables
  1513→    if (matchesPrefix(num, payablesPrefix) || (!payablesPrefix && hasKeyword(n, PAYABLES_KEYWORDS))) {
  1514→      payables += creditBal;
  1515→    }
  1516→    // Short-term debt
  1517→    if (matchesPrefix(num, shortTermDebtPrefix) || (!shortTermDebtPrefix && hasKeyword(n, SHORT_TERM_DEBT_KEYWORDS))) {
  1518→      shortTermDebt += creditBal;
  1519→    }
  1520→    // Long-term debt
  1521→    if (matchesPrefix(num, longTermDebtPrefix) || (!longTermDebtPrefix && hasKeyword(n, LONG_TERM_DEBT_KEYWORDS))) {
  1522→      longTermDebt += creditBal;
  1523→    }
  1524→
  1525→    // ── Main category detection by prefix ──
  1526→    if (matchesPrefix(num, currentAssetsPrefix)) {
  1527→      currentAssets += debitBal;
  1528→    } else if (matchesPrefix(num, nonCurrentAssetsPrefix)) {
  1529→      nonCurrentAssets += debitBal;
  1530→    } else if (matchesPrefix(num, currentLiabPrefix)) {
  1531→      currentLiabilities += creditBal;
  1532→    } else if (matchesPrefix(num, nonCurrentLiabPrefix)) {
  1533→      nonCurrentLiabilities += creditBal;
  1534→    } else if (matchesPrefix(num, equityPrefix)) {
  1535→      equity += creditBal;
  1536→    } else if (num.indexOf("1") === 0) {
  1537→      currentAssets += debitBal;
  1538→    } else if (num.indexOf("2") === 0) {
  1539→      currentLiabilities += creditBal;
  1540→    } else if (num.indexOf("3") === 0) {
  1541→      equity += creditBal;
  1542→    }
  1543→  }
  1544→
  1545→  const totalAssets = r2(currentAssets + nonCurrentAssets);
  1546→  const totalLiabilities = r2(currentLiabilities + nonCurrentLiabilities);
  1547→  const totalDebt = r2(shortTermDebt + longTermDebt);
  1548→
  1549→  return {
  1550→    currentAssets: r2(currentAssets),
  1551→    nonCurrentAssets: r2(nonCurrentAssets),
  1552→    totalAssets,
  1553→    currentLiabilities: r2(currentLiabilities),
  1554→    nonCurrentLiabilities: r2(nonCurrentLiabilities),
  1555→    totalLiabilities,
  1556→    equity: r2(equity),
  1557→    inventory: r2(inventory),
  1558→    cash: r2(cash),
  1559→    receivables: r2(receivables),
  1560→    fixedAssets: r2(fixedAssets),
  1561→    payables: r2(payables),
  1562→    shortTermDebt: r2(shortTermDebt),
  1563→    longTermDebt: r2(longTermDebt),
  1564→    totalDebt,
  1565→  };
  1566→}

  1605→/** Convert a turnover ratio (e.g. inventoryTurnover) to days (365 / turnover). Returns null if input is null/0.

  1609→  return r2(365 / turnover);

  1821→export function fmtRatio(v: number | null, unit: "ratio" | "percent" | "amount" | "days"): string {
  1822→  if (v == null) return "—";
  1823→  if (unit === "percent") return (v * 100).toFixed(1) + "%";
  1824→  if (unit === "amount") return v.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  1825→  if (unit === "days") return v.toFixed(0) + " يوم";
  1826→  return v.toFixed(2) + "×";
