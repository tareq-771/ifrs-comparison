   126→const AR_DIGITS = "٠١٢٣٤٥٦٧٨٩";

   128→export function normalizeName(v: unknown): string {
   129→  let t = String(v ?? "").normalize("NFKC");
   130→  t = t.replace(/[\u200B-\u200F\uFEFF]/g, "");
   131→  t = t.replace(/[\u064B-\u065F\u0670\u0640]/g, "");
   132→  t = t.replace(/[^\w\u0600-\u06FF\s]/g, " ");
   133→  t = t.replace(/\s+/g, " ").trim().toLowerCase();
   134→  t = t.replace(/ة/g, "ه").replace(/ى/g, "ي").replace(/[أإآ]/g, "ا");
   135→  return t;
   136→}

   138→export function parseAmount(v: unknown): number {
   139→  if (v == null || v === "") return 0;
   140→  if (typeof v === "number") return v;
   141→  let s = String(v)
   142→    .replace(/[٠-٩]/g, (d) => String(AR_DIGITS.indexOf(d)))
   143→    .replace(/[,،\s]/g, "");
   144→  const neg = s[0] === "-";
   145→  s = s.replace(/[-*]+/g, "");
   146→  const x = parseFloat(s);
   147→  return Number.isNaN(x) ? 0 : neg ? -x : x;
   148→}

   150→export function parseAccountNum(v: unknown): string | null {
   151→  if (v == null || v === "") return null;
   152→  if (typeof v === "number") return String(Math.round(v));
   153→  const s = String(v)
   154→    .replace(/[٠-٩]/g, (d) => String(AR_DIGITS.indexOf(d)))
   155→    .trim();
   156→  const d = s.replace(/\D/g, "");
   157→  return d ? d.replace(/^0+(?=.)/, "") || "0" : null;
   158→}

   164→export function readExcelFile(file: File): Promise<FileData> {
   165→  return new Promise((resolve, reject) => {
   166→    const fr = new FileReader();
   167→    fr.onload = (e) => {
   168→      try {
   169→        const buf = e.target?.result;
   170→        if (!buf) throw new Error("ملف فارغ");
   171→        const wb = XLSX.read(buf, { type: "array" });
   172→        const sh = wb.SheetNames[0];
   173→        const rows = XLSX.utils.sheet_to_json<unknown[]>(wb.Sheets[sh], {
   174→          header: 1, raw: true, defval: null,
   175→        });
   176→        let hi = -1;
   177→        const cap = Math.min(rows.length, 30);
   178→        for (let i = 0; i < cap; i++) {
   179→          const cells = (rows[i] || []).map((c) => String(c ?? "").trim());
   180→          for (let j = 0; j < cells.length; j++) {
   181→            if (cells[j].indexOf("اسم الحساب") >= 0) { hi = i; break; }
   182→          }
   183→          if (hi >= 0) break;
   184→        }
   185→        if (hi < 0) throw new Error("لا يوجد عمود «اسم الحساب»");
   186→        const hdr = rows[hi].map((c) => String(c ?? "").trim());
   187→        const sub = (rows[hi + 1] || []).map((c) => String(c ?? "").trim());
   188→        let cN = 1, cD = -1, cNum = -1;
   189→        for (let j = 0; j < hdr.length; j++) {
   190→          const h = hdr[j];
   191→          if (h.indexOf("اسم الحساب") >= 0) cN = j;
   192→          if (sub[j] === "مدين" || h === "مدين") cD = j;
   193→        }
   194→        // Detect account number column AFTER name column — avoid picking "اسم الحساب"
   195→        for (let j = 0; j < hdr.length; j++) {
   196→          const h = hdr[j];
   197→          if (j === cN) continue; // skip the name column
   198→          if (
   199→            h.indexOf("رقم الحساب") >= 0 ||
   200→            h.indexOf("رقم") >= 0 ||
   201→            h === "الحساب" ||
   202→            h === "الرقم" ||
   203→            h === "حساب"
   204→          ) { cNum = j; break; }
   205→        }
   206→        if (cNum < 0) cNum = 0; // fallback: column 0
   207→        if (cD < 0) {
   208→          for (let j = 0; j < hdr.length; j++) {
   209→            if (hdr[j] === "المبلغ") { cD = j; break; }
   210→          }
   211→        }
   212→        if (cD < 0) cD = cN + 2;
   213→        const A: AccountRow[] = [];
   214→        for (let i = hi + 1; i < rows.length; i++) {
   215→          const r = rows[i] || [];
   216→          const g = (j: number) => (j >= 0 && j < r.length ? r[j] : null);
   217→          const n = String(g(cN) ?? "").trim();
   218→          if (!n || n === "*") continue;
   219→          if (n.indexOf("الإجمالي") >= 0 || n.indexOf("الرصيد") >= 0) continue;
   220→          A.push({ nm: n, nk: normalizeName(n), num: parseAccountNum(g(cNum)), m: parseAmount(g(cD)), d: parseAmount(g(cD + 1)) });
   221→        }
   222→        resolve({ A, headers: hdr, nameCol: cN, numCol: cNum, debitCol: cD, headerRow: hi, rawRows: rows });
   223→      } catch (err) {
   224→        reject(err instanceof Error ? err : new Error(String(err)));
   225→      }
   226→    };
   227→    fr.onerror = () => reject(new Error("فشل قراءة الملف"));
   228→    fr.readAsArrayBuffer(file);
   229→  });
   230→}

   232→export function reExtract(file: FileData, nameCol: number, debitCol: number): AccountRow[] {
   233→  const rows = file.rawRows;
   234→  const hi = file.headerRow;
   235→  const numCol = file.numCol ?? 0;
   236→  const A: AccountRow[] = [];
   237→  for (let i = hi + 1; i < rows.length; i++) {
   238→    const r = rows[i] || [];
   239→    const g = (j: number) => (j >= 0 && j < r.length ? r[j] : null);
   240→    const n = String(g(nameCol) ?? "").trim();
   241→    if (!n || n === "*") continue;
   242→    if (n.indexOf("الإجمالي") >= 0 || n.indexOf("الرصيد") >= 0) continue;
   243→    A.push({ nm: n, nk: normalizeName(n), num: parseAccountNum(g(numCol)), m: parseAmount(g(debitCol)), d: parseAmount(g(debitCol + 1)) });
   244→  }
   245→  return A;
   246→}

   252→export function matchFiles(f1: FileData, f2: FileData): MatchedRow[] {
   253→  const out: MatchedRow[] = [];
   254→  const m2: Record<string, AccountRow[]> = {};
   255→  for (const a of f2.A) { (m2[a.nk] ||= []).push(a); }
   256→  for (const a1 of f1.A) {
   257→    const q = m2[a1.nk];
   258→    out.push({ nk: a1.nk, a1, a2: q && q.length ? q.shift()! : null, st: "مطابق" });
   259→  }
   260→  for (const k in m2) for (const a2 of m2[k]) out.push({ nk: k, a1: null, a2, st: "مطابق" });
   261→  for (const r of out) {
   262→    r.st = r.a1 && r.a2 ? "مطابق" : r.a1 ? "في الملف الأول فقط" : "في الملف الثاني فقط";
   263→  }
   264→  return out;
   265→}

   268→ * Chart-of-Accounts Matching — مطابقة دليل الحسابات
   269→ *
   270→ * Matches accounts by NAME AND NUMBER (OR logic):
   271→ *  - Name match   — normalized name similarity >= threshold (default 90%)
   272→ *  - Number match — account numbers are identical (exact match)
   273→ *
   274→ * Priority: both > number-only > name-only.
   275→ * Statuses: مطابق / تغير اسم / تغير رقم / تغير الموقع / في الملف الأول فقط / في الملف الثاني فقط
   276→ * ════════════════════════════════════════════════════════════════════════════ */

   278→/** Levenshtein edit distance between two strings (classic DP). */
   279→function levenshtein(a: string, b: string): number {
   280→  if (a === b) return 0;
   281→  if (!a.length) return b.length;
   282→  if (!b.length) return a.length;
   283→  let prev = new Array<number>(b.length + 1);
   284→  let curr = new Array<number>(b.length + 1);
   285→  for (let j = 0; j <= b.length; j++) prev[j] = j;
   286→  for (let i = 1; i <= a.length; i++) {
   287→    curr[0] = i;
   288→    for (let j = 1; j <= b.length; j++) {
   289→      const cost = a.charCodeAt(i - 1) === b.charCodeAt(j - 1) ? 0 : 1;
   290→      curr[j] = Math.min(curr[j - 1] + 1, prev[j] + 1, prev[j - 1] + cost);
   291→    }
   292→    const tmp = prev; prev = curr; curr = tmp;
   293→  }
   294→  return prev[b.length];
   295→}

   297→/** Name similarity ratio in [0, 1]. Levenshtein normalized by max length. */
   298→export function nameSimilarity(a: string, b: string): number {
   299→  if (!a && !b) return 0;
   300→  if (!a || !b) return 0;
   301→  if (a === b) return 1;
   302→  const maxLen = Math.max(a.length, b.length);
   303→  if (maxLen === 0) return 0;
   304→  return 1 - levenshtein(a, b) / maxLen;
   305→}

   307→/** Character-level LCS-based diff. Returns segments of {text, op}. */
   308→export type DiffOp = "same" | "add" | "del";
   309→export interface DiffSegment { text: string; op: DiffOp; }
   310→
   311→export function diffStrings(a: string, b: string): DiffSegment[] {
   312→  if (!a && !b) return [];
   313→  if (!a) return [{ text: b, op: "add" }];
   314→  if (!b) return [{ text: a, op: "del" }];
   315→  if (a === b) return [{ text: a, op: "same" }];
   316→  const m = a.length, n = b.length;
   317→  const dp: number[][] = Array.from({ length: m + 1 }, () => new Array<number>(n + 1).fill(0));
   318→  for (let i = 1; i <= m; i++) {
   319→    for (let j = 1; j <= n; j++) {
   320→      if (a.charCodeAt(i - 1) === b.charCodeAt(j - 1)) dp[i][j] = dp[i - 1][j - 1] + 1;
   321→      else dp[i][j] = Math.max(dp[i - 1][j], dp[i][j - 1]);
   322→    }
   323→  }
   324→  const segs: DiffSegment[] = [];
   325→  let i = m, j = n;
   326→  while (i > 0 || j > 0) {
   327→    if (i > 0 && j > 0 && a.charCodeAt(i - 1) === b.charCodeAt(j - 1)) {
   328→      segs.push({ text: a[i - 1], op: "same" }); i--; j--;
   329→    } else if (j > 0 && (i === 0 || dp[i][j - 1] >= dp[i - 1][j])) {
   330→      segs.push({ text: b[j - 1], op: "add" }); j--;
   331→    } else {
   332→      segs.push({ text: a[i - 1], op: "del" }); i--;
   333→    }
   334→  }
   335→  segs.reverse();
   336→  const merged: DiffSegment[] = [];
   337→  for (const s of segs) {
   338→    const last = merged[merged.length - 1];
   339→    if (last && last.op === s.op) last.text += s.text;
   340→    else merged.push({ ...s });
   341→  }
   342→  return merged;
   343→}

   345→/** Compact human-readable diff summary for Excel: "−X +Y" or "مطابق تام". */
   346→export function summarizeDiff(a: string, b: string): string {
   347→  if (!a || !b) return "—";
   348→  if (a === b) return "مطابق تام";
   349→  const diff = diffStrings(a, b);
   350→  let removed = "";
   351→  let added = "";
   352→  for (const s of diff) {
   353→    if (s.op === "del") removed += s.text;
   354→    else if (s.op === "add") added += s.text;
   355→  }
   356→  const parts: string[] = [];
   357→  if (removed) parts.push(`−${removed}`);
   358→  if (added) parts.push(`+${added}`);
   359→  return parts.length ? parts.join(" ") : "مطابق تام";
   360→}

   382→    case "5":

   385→    case "7": return "إيرادات";

   390→/**
   391→ * Match two charts of accounts by NAME AND NUMBER (OR logic).
   392→ * A pair is a candidate if EITHER number matches exactly OR name sim >= threshold.
   393→ * Priority: both > number-only > name-only.
   394→ */
   395→export function matchChartOfAccounts(
   396→  f1: FileData,
   397→  f2: FileData,
   398→  threshold = 0.9
   399→): MatchedRow[] {
   400→  const a1s = f1.A.slice();
   401→  const a2s = f2.A.slice();
   402→  type Pair = { i: number; j: number; sim: number; numMatch: boolean; nameMatch: boolean };
   403→  const pairs: Pair[] = [];
   404→  for (let i = 0; i < a1s.length; i++) {
   405→    const a1 = a1s[i];
   406→    if (!a1.nk && !a1.num) continue;
   407→    for (let j = 0; j < a2s.length; j++) {
   408→      const a2 = a2s[j];
   409→      if (!a2.nk && !a2.num) continue;
   410→      const numMatch = !!a1.num && !!a2.num && a1.num === a2.num;
   411→      const sim = nameSimilarity(a1.nk, a2.nk);
   412→      const nameMatch = sim >= threshold;
   413→      if (numMatch || nameMatch) {
   414→        pairs.push({ i, j, sim, numMatch, nameMatch });
   415→      }
   416→    }
   417→  }
   418→  function pairPriority(p: Pair): number {
   419→    if (p.numMatch && p.nameMatch) return 3;
   420→    if (p.numMatch) return 2;
   421→    return 1;
   422→  }
   423→  pairs.sort((x, y) => {
   424→    const px = pairPriority(x);
   425→    const py = pairPriority(y);
   426→    if (px !== py) return py - px;
   427→    return y.sim - x.sim;
   428→  });
   429→
   430→  const used1 = new Set<number>();
   431→  const used2 = new Set<number>();
   432→  const out: MatchedRow[] = [];
   433→
   434→  for (const p of pairs) {
   435→    if (used1.has(p.i) || used2.has(p.j)) continue;
   436→    used1.add(p.i);
   437→    used2.add(p.j);
   438→    const a1 = a1s[p.i];
   439→    const a2 = a2s[p.j];
   440→    const cat1 = accountCategory(a1.num);
   441→    const cat2 = accountCategory(a2.num);
   442→    const numChanged = (a1.num ?? null) !== (a2.num ?? null);
   443→    const catChanged = cat1 !== cat2;
   444→    const nameChanged = a1.nk !== a2.nk;
   445→    let matchBy: "both" | "name" | "number";
   446→    if (p.numMatch && p.nameMatch) matchBy = "both";
   447→    else if (p.numMatch) matchBy = "number";
   448→    else matchBy = "name";
   449→
   450→    // ── 4-sheet classification (per user's new requirement) ──
   451→    // Sheet 1 «مطابق الاسم والرقم»:   name AND number both match (priority to name)
   452→    // Sheet 2 «مطابق الاسم مختلف الرقم»: name matches, number differs (any category)
   453→    // Sheet 3 «غير مطابق الاسم»:        unmatched (in one file only) — handled below
   454→    // Sheet 4 «التغير الجوهري»:         category CHANGED between the two files
   455→    //                                    (e.g. account moved from أصول to خصوم,
   456→    //                                     or from مصروفات to إيرادات)
   457→    let st: MatchStatus;
   458→    if (catChanged) {
   459→      // Category change is the fundamental change → Sheet 4
   460→      st = "تغير جوهري";
   461→    } else if (!numChanged && !nameChanged) {
   462→      // Both name AND number match → Sheet 1
   463→      st = "مطابق";
   464→    } else if (!nameChanged && numChanged) {
   465→      // Name matches, number differs, same category → Sheet 2
   466→      st = "مطابق الاسم مختلف الرقم";
   467→    } else {
   468→      // nameChanged && numChanged (both differ) — matched by one criterion only
   469→      // Same category, but both name AND number differ
   470→      // → keep in Sheet 2 if matched by name (name is the priority criterion)
   471→      // → otherwise keep in Sheet 2 too (the user wants name-priority matching)
   472→      st = "مطابق الاسم مختلف الرقم";
   473→    }
   474→    // Compute amount change + change percentage (debit−credit balance)
   475→    const amt1 = a1.m - a1.d;
   476→    const amt2 = a2.m - a2.d;
   477→    const amtChange = amt2 - amt1;
   478→    const amtChangePct = amt1 !== 0 ? (amtChange / Math.abs(amt1)) * 100 : null;
   479→    out.push({
   480→      nk: a1.nk, a1, a2, st,
   481→      sim: Math.round(p.sim * 100),
   482→      cat1, cat2,
   483→      numChanged, catChanged, nameChanged, matchBy,
   484→      amt1, amt2, amtChange, amtChangePct,
   485→    });
   486→  }
   487→
   488→  for (let i = 0; i < a1s.length; i++) {
   489→    if (used1.has(i)) continue;
   490→    const a1 = a1s[i];
   491→    out.push({
   492→      nk: a1.nk, a1, a2: null, st: "في الملف الأول فقط",
   493→      sim: null, cat1: accountCategory(a1.num), cat2: undefined,
   494→      numChanged: false, catChanged: false, nameChanged: false, matchBy: undefined,
   495→      amt1: a1.m - a1.d, amt2: null, amtChange: null, amtChangePct: null,
   496→    });
   497→  }
   498→  for (let j = 0; j < a2s.length; j++) {
   499→    if (used2.has(j)) continue;
   500→    const a2 = a2s[j];
   501→    out.push({
   502→      nk: a2.nk, a1: null, a2, st: "في الملف الثاني فقط",
   503→      sim: null, cat1: undefined, cat2: accountCategory(a2.num),
   504→      numChanged: false, catChanged: false, nameChanged: false, matchBy: undefined,
   505→      amt1: null, amt2: a2.m - a2.d, amtChange: null, amtChangePct: null,
   506→    });
   507→  }
   508→
   509→  // Sort: matched → name-match-num-diff → fundamental → unmatched
   510→  const order: Record<MatchStatus, number> = {
   511→    "مطابق": 0,
   512→    "مطابق الاسم مختلف الرقم": 1,
   513→    "تغير جوهري": 2,
   514→    "في الملف الأول فقط": 3,
   515→    "في الملف الثاني فقط": 4,
   516→    "غير مطابق": 5,
   517→  };
   518→  out.sort((x, y) => {
   519→    const o = order[x.st] - order[y.st];
   520→    if (o !== 0) return o;
   521→    const sx = x.sim ?? -1;
   522→    const sy = y.sim ?? -1;
   523→    return sy - sx;
   524→  });
   525→  return out;
   526→}

   532→export interface BaseOption { nk: string; nm: string; sc: number; }

   534→export function collectBaseOptions(files: (FileData | null)[]): BaseOption[] {
   535→  const seen: Record<string, true> = {};
   536→  const opts: BaseOption[] = [];
   537→  for (const d of files) {

   543→      const sc = x.indexOf("صافي") >= 0 && x.indexOf("مبيعات") >= 0 ? 0 : x.indexOf("مبيعات") >= 0 ? 1 : 2;
   544→      opts.push({ nk: a.nk, nm: a.nm, sc });
   545→    }

   548→  return opts;

   551→/* ──────────────────────────────────────────────────────────────────────────
   552→ * Categorization — prefix-based (IAS 1 function-of-expense)
   553→ * ────────────────────────────────────────────────────────────────────────── */
   554→
   555→export function categorize(out: MatchedRow[], settings: CategorySettings): Categorized {
   556→  const { base, costPrefix, sellPrefix, adminPrefix, otherOpPrefix, financePrefix, taxPrefix, ociPrefix } = settings;
   557→
   558→  const ba = base ? out.find((r) => r.nk === base) ?? null : null;
   559→  const netRow = findNetRow(out);
   560→  const baseNum = (ba?.a1?.num ?? ba?.a2?.num ?? null) ?? null;
   561→
   562→  const cat: Categorized = {
   563→    sales: [], cost: [], sell: [], adm: [], oth: [], fin: [], tax: [], oci: [], othRev: [], net: [],
   564→  };
   565→
   566→  for (const r of out) {
   567→    // Exclude base and net-income summary rows from categorization
   568→    if (r === ba || r === netRow) continue;
   569→
   570→    const num = String((r.a1?.num ?? r.a2?.num ?? "") as string);
   571→
   572→    // Revenue tree — accounts whose number starts with base number
   573→    if (baseNum && num && num.indexOf(baseNum) === 0) { cat.sales.push(r); continue; }
   574→    // Cost of sales
   575→    if (costPrefix && num && num.indexOf(costPrefix) === 0) { cat.cost.push(r); continue; }
   576→    // Selling & distribution
   577→    if (sellPrefix && num && num.indexOf(sellPrefix) === 0) { cat.sell.push(r); continue; }
   578→    // General & administrative
   579→    if (adminPrefix && num && num.indexOf(adminPrefix) === 0) { cat.adm.push(r); continue; }
   580→    // Other operating
   581→    if (otherOpPrefix && num && num.indexOf(otherOpPrefix) === 0) { cat.oth.push(r); continue; }
   582→    // Finance
   583→    if (financePrefix && num && num.indexOf(financePrefix) === 0) { cat.fin.push(r); continue; }
   584→    // Tax
   585→    if (taxPrefix && num && num.indexOf(taxPrefix) === 0) { cat.tax.push(r); continue; }
   586→    // OCI
   587→    if (ociPrefix && num && num.indexOf(ociPrefix) === 0) { cat.oci.push(r); continue; }
   588→    // Unprefixed accounts starting with "3" → other operating
   589→    if (num && num.indexOf("3") === 0) { cat.oth.push(r); continue; }
   590→    // Everything else → other income
   591→    cat.othRev.push(r);
   592→  }
   593→
   594→  const byNum = (a: MatchedRow, b: MatchedRow) => {
   595→    const x = String((a.a1?.num ?? a.a2?.num ?? "") as string);
   596→    const y = String((b.a1?.num ?? b.a2?.num ?? "") as string);
   597→    return x < y ? -1 : x > y ? 1 : 0;
   598→  };
   599→  cat.sales.sort(byNum); cat.cost.sort(byNum); cat.sell.sort(byNum);
   600→  cat.adm.sort(byNum); cat.oth.sort(byNum); cat.fin.sort(byNum);
   601→  cat.tax.sort(byNum); cat.oci.sort(byNum); cat.othRev.sort(byNum);
   602→  return cat;
   603→}

   605→/** Find the "صافي الدخل" summary row (for reconciliation). */
   606→export function findNetRow(out: MatchedRow[]): MatchedRow | null {
   607→  for (const r of out) {
   608→    const n = normalizeName((r.a1 ?? r.a2)?.nm ?? "");
   609→    if (n.indexOf("صافي الدخل") >= 0 || n.indexOf("صافي دخ") >= 0) return r;
   610→  }
   611→  return null;
   612→}

   614→/* ──────────────────────────────────────────────────────────────────────────
   615→ * Leaf detection — parents are shown bold but excluded from totals
   616→ * ────────────────────────────────────────────────────────────────────────── */
   617→
   618→export interface RowLeafInfo {
   619→  leaf1: boolean;
   620→  leaf2: boolean;
   621→}
   622→
   623→export function leafInfo(rows: MatchedRow[]): RowLeafInfo[] {
   624→  return rows.map((r, i) => {
   625→    let leaf1 = false, leaf2 = false;
   626→    const a1 = r.a1;
   627→    if (a1 && a1.num) {
   628→      leaf1 = true;
   629→      for (let j = 0; j < rows.length; j++) {
   630→        if (j === i) continue;
   631→        const o = rows[j].a1;
   632→        if (!o || !o.num) continue;
   633→        if (o.num.length > a1.num.length && o.num.indexOf(a1.num) === 0) { leaf1 = false; break; }
   634→      }
   635→    }
   636→    const a2 = r.a2;
   637→    if (a2 && a2.num) {
   638→      leaf2 = true;
   639→      for (let j = 0; j < rows.length; j++) {
   640→        if (j === i) continue;
   641→        const o = rows[j].a2;
   642→        if (!o || !o.num) continue;
   643→        if (o.num.length > a2.num.length && o.num.indexOf(a2.num) === 0) { leaf2 = false; break; }
   644→      }
   645→    }
   646→    return { leaf1, leaf2 };
   647→  });
   648→}

   650→/* ──────────────────────────────────────────────────────────────────────────
   651→ * Totals — IFRS cascade with credit/debit sign convention
   652→ * ────────────────────────────────────────────────────────────────────────── */
   653→
   654→const r2 = (n: number) => Math.round(n * 100) / 100;
   655→
   656→/**
   657→ * Sum LEAF accounts of a category on one file side.
   658→ * @param isRevenue  true → credit-positive (دائن − مدين), false → debit-positive (مدين − دائن)
   659→ */
   660→function sumLeaves(
   661→  rows: MatchedRow[], info: RowLeafInfo[], side: "a1" | "a2", isRevenue: boolean
   662→): number {
   663→  let s = 0;
   664→  for (let i = 0; i < rows.length; i++) {
   665→    const isLeaf = side === "a1" ? info[i].leaf1 : info[i].leaf2;
   666→    if (!isLeaf) continue;
   667→    const a = rows[i][side]!;
   668→    s += isRevenue ? a.d - a.m : a.m - a.d;
   669→  }
   670→  return r2(s);
   671→}

   673→/**
   674→ * Compute all income-statement totals.
   675→ * @param baseRow  The selected net-sales account (its OWN balance = revenue).
   676→ */
   677→export function calcTotals(
   678→  cat: Categorized,
   679→  baseRow: MatchedRow | null
   680→): Totals {
   681→  // Revenue = base account's own balance (credit − debit)
   682→  const rev1 = baseRow?.a1 ? r2(baseRow.a1.d - baseRow.a1.m) : 0;
   683→  const rev2 = baseRow?.a2 ? r2(baseRow.a2.d - baseRow.a2.m) : 0;
   684→
   685→  const iCost = leafInfo(cat.cost);
   686→  const iSell = leafInfo(cat.sell);
   687→  const iAdm = leafInfo(cat.adm);
   688→  const iOth = leafInfo(cat.oth);
   689→  const iFin = leafInfo(cat.fin);
   690→  const iTax = leafInfo(cat.tax);
   691→  const iOci = leafInfo(cat.oci);
   692→  const iOthRev = leafInfo(cat.othRev);
   693→
   694→  const cost1 = sumLeaves(cat.cost, iCost, "a1", false);
   695→  const cost2 = sumLeaves(cat.cost, iCost, "a2", false);
   696→  const sell1 = sumLeaves(cat.sell, iSell, "a1", false);
   697→  const sell2 = sumLeaves(cat.sell, iSell, "a2", false);
   698→  const adm1 = sumLeaves(cat.adm, iAdm, "a1", false);
   699→  const adm2 = sumLeaves(cat.adm, iAdm, "a2", false);
   700→  const oth1 = sumLeaves(cat.oth, iOth, "a1", false);
   701→  const oth2 = sumLeaves(cat.oth, iOth, "a2", false);
   702→  const othRev1 = sumLeaves(cat.othRev, iOthRev, "a1", true);
   703→  const othRev2 = sumLeaves(cat.othRev, iOthRev, "a2", true);
   704→  const fin1 = sumLeaves(cat.fin, iFin, "a1", false);
   705→  const fin2 = sumLeaves(cat.fin, iFin, "a2", false);
   706→  const tax1 = sumLeaves(cat.tax, iTax, "a1", false);
   707→  const tax2 = sumLeaves(cat.tax, iTax, "a2", false);
   708→  const oci1 = sumLeaves(cat.oci, iOci, "a1", true);
   709→  const oci2 = sumLeaves(cat.oci, iOci, "a2", true);
   710→
   711→  const gross1 = r2(rev1 - cost1);
   712→  const gross2 = r2(rev2 - cost2);
   713→  const opex1 = r2(sell1 + adm1 + oth1);
   714→  const opex2 = r2(sell2 + adm2 + oth2);
   715→  const oper1 = r2(gross1 - opex1);
   716→  const oper2 = r2(gross2 - opex2);
   717→  const pbt1 = r2(oper1 + othRev1 - fin1);
   718→  const pbt2 = r2(oper2 + othRev2 - fin2);
   719→  const net1 = r2(pbt1 - tax1);
   720→  const net2 = r2(pbt2 - tax2);
   721→  const comp1 = r2(net1 + oci1);
   722→  const comp2 = r2(net2 + oci2);
   723→
   724→  return {
   725→    rev1, rev2, cost1, cost2, gross1, gross2,
   726→    sell1, sell2, adm1, adm2, oth1, oth2, opex1, opex2,
   727→    oper1, oper2, othRev1, othRev2, fin1, fin2,
   728→    pbt1, pbt2, tax1, tax2, net1, net2, oci1, oci2, comp1, comp2,
   729→  };
   730→}

   759→/* ──────────────────────────────────────────────────────────────────────────
   760→ * Change analysis — desirability & percentage
   761→ * ──────────────────────────────────────────────────────────────────────────
   762→ *
   763→ * For EXPENSE accounts (prefix 3, debit-positive):
   764→ *   - Decrease (chg < 0) = savings = مرغوب  (desirable)
   765→ *   - Increase (chg > 0) = overspend = غير مرغوب (undesirable)
   766→ *
   767→ * For REVENUE accounts (prefix 4, credit-positive):
   768→ *   - Increase (chg > 0) = more income = مرغوب  (desirable)
   769→ *   - Decrease (chg < 0) = lost income = غير مرغوب (undesirable)
   770→ *
   771→ * Change percentage = chg / base period (v1 = comparative/older period)
   772→ */
   773→
   774→export type ChangeStatus =
   775→  | "مرغوب نمو"
   776→  | "غير مرغوب انخفاض"
   777→  | "مرغوب وفرة"
   778→  | "غير مرغوب زيادة"
   779→  | "—";
   780→
   781→export function computeChangeStatus(
   782→  chg: number | null,
   783→  isRevenue: boolean
   784→): ChangeStatus {
   785→  if (chg == null || chg === 0) return "—";
   786→  if (isRevenue) {
   787→    // Revenue: increase = good (growth), decrease = bad (decline)
   788→    return chg > 0 ? "مرغوب نمو" : "غير مرغوب انخفاض";
   789→  }
   790→  // Expense: decrease = good (savings), increase = bad (overspend)
   791→  return chg < 0 ? "مرغوب وفرة" : "غير مرغوب زيادة";
   792→}
   793→
   794→export function computeChangePct(
   795→  chg: number | null,
   796→  v1: number | null
   797→): number | null {
   798→  if (chg == null || v1 == null || v1 === 0) return null;
   799→  return chg / v1;
   800→}

   802→/* ──────────────────────────────────────────────────────────────────────────
   803→ * Comparison modes
   804→ * ──────────────────────────────────────────────────────────────────────────
   805→ *
   806→ * Mode 1 — "period": Current period vs Previous period (change analysis)
   807→ * Mode 2 — "monthCumulative": Current month vs Year-to-date cumulative
   808→ *          (month's contribution to the annual total)
   809→ *
   810→ * In mode 2, the key metric is monthRatio = month_value / cumulative_value.
   811→ * A "normal" month should represent ~1/numMonths of the annual total.
   812→ * Desirability:
   813→ *   Revenue: month ratio > expected = مرغوب (strong month)
   814→ *   Expense: month ratio > expected = غير مرغوب (high expense month)
   815→ */
   816→
   817→export type CompareMode = "period" | "monthCumulative" | "chart";
   818→
   819→/** Month/cumulative ratio = month value (v2) / cumulative value (v1). */
   820→export function computeMonthRatio(
   821→  v2: number | null,
   822→  v1: number | null
   823→): number | null {
   824→  if (v2 == null || v1 == null || v1 === 0) return null;
   825→  return v2 / v1;
   826→}
   827→
   828→/** Remaining = cumulative − month (what other months contributed). */
   829→export function computeRemaining(
   830→  v1: number | null,
   831→  v2: number | null
   832→): number | null {
   833→  if (v1 == null || v2 == null) return null;
   834→  return r2(v1 - v2);
   835→}
   836→
   837→/** Average cumulative = cumulative value / number of months. */
   838→export function computeAvgCumulative(
   839→  v1: number | null,
   840→  numMonths: number
   841→): number | null {
   842→  if (v1 == null || numMonths === 0) return null;
   843→  return r2(v1 / numMonths);
   844→}
   845→
   846→/**
   847→ * Ratio of month value to the cumulative average.
   848→ * month / (cumulative / numMonths) = month × numMonths / cumulative.
   849→ * Values > 1 mean the month exceeds the average.
   850→ */
   851→export function computeMonthToAvgRatio(
   852→  v2: number | null,
   853→  avgCumulative: number | null
   854→): number | null {
   855→  if (v2 == null || avgCumulative == null || avgCumulative === 0) return null;
   856→  return v2 / avgCumulative;
   857→}
   858→
   859→/** Absolute difference: month value − average cumulative. */
   860→export function computeMonthAvgDiff(
   861→  v2: number | null,
   862→  avgCumulative: number | null
   863→): number | null {
   864→  if (v2 == null || avgCumulative == null) return null;
   865→  return r2(v2 - avgCumulative);
   866→}
   867→
   868→/** Percentage difference from average: (month − avg) / avg. */
   869→export function computeMonthAvgDiffPct(
   870→  v2: number | null,
   871→  avgCumulative: number | null
   872→): number | null {
   873→  if (v2 == null || avgCumulative == null || avgCumulative === 0) return null;
   874→  return (v2 - avgCumulative) / avgCumulative;
   875→}
   876→
   877→/**
   878→ * Desirability of the month's concentration vs the expected average.
   879→ * @param monthRatio  month/cumulative ratio (0..1)
   880→ * @param isRevenue   true for revenue, false for expense
   881→ * @param numMonths   number of months in the cumulative period (default 12)
   882→ */
   883→export function computeMonthStatus(
   884→  monthRatio: number | null,
   885→  isRevenue: boolean,
   886→  numMonths = 12
   887→): ChangeStatus {
   888→  if (monthRatio == null || monthRatio === 0) return "—";
   889→  const expected = 1 / numMonths;
   890→  if (Math.abs(monthRatio - expected) < 0.005) return "—";
   891→  if (isRevenue) {
   892→    return monthRatio > expected ? "مرغوب نمو" : "غير مرغوب انخفاض";
   893→  }
   894→  return monthRatio > expected ? "غير مرغوب زيادة" : "مرغوب وفرة";
   895→}

  1447→/**
  1448→ * Parse a balance-sheet file and compute category totals.
  1449→ * Uses the same parsing as income-statement files (readExcelFile).
  1450→ * The `totals` are computed from LEAF accounts only (same logic as income statement).
  1451→ *
  1452→ * Detailed items (inventory, cash, etc.) are detected by PREFIX first;
  1453→ * if the prefix is empty, falls back to keyword-based name detection.
  1454→ */
  1455→export function categorizeBalanceSheet(
  1456→  file: FileData,
  1457→  settings: BalanceSheetSettings
  1458→): BalanceSheetTotals {
  1459→  const {
  1460→    currentAssetsPrefix, nonCurrentAssetsPrefix, currentLiabPrefix, nonCurrentLiabPrefix, equityPrefix,
  1461→    inventoryPrefix, cashPrefix, receivablesPrefix, fixedAssetsPrefix,
  1462→    payablesPrefix, shortTermDebtPrefix, longTermDebtPrefix,
  1463→  } = settings;
  1464→
  1465→  // Mark leaves
  1466→  const nums = file.A.map((a) => a.num).filter(Boolean) as string[];
  1467→  for (const a of file.A) {
  1468→    if (!a.num) { (a as any).leaf = true; continue; }
  1469→    (a as any).leaf = true;
  1470→    for (const n of nums) {
  1471→      if (n !== a.num && n.length > a.num.length && n.indexOf(a.num) === 0) {
  1472→        (a as any).leaf = false;
  1473→        break;
  1474→      }
  1475→    }
  1476→  }
  1477→
  1478→  let currentAssets = 0, nonCurrentAssets = 0;
  1479→  let currentLiabilities = 0, nonCurrentLiabilities = 0;
  1480→  let equity = 0;
  1481→  let inventory = 0, cash = 0, receivables = 0;
  1482→  let fixedAssets = 0, payables = 0, shortTermDebt = 0, longTermDebt = 0;
  1483→
  1484→  for (const a of file.A) {
  1485→    if (!(a as any).leaf) continue;
  1486→    const num = a.num ?? "";
  1487→    const n = normalizeName(a.nm);
  1488→    const debitBal = a.m - a.d;   // asset = debit − credit
  1489→    const creditBal = a.d - a.m;  // liability/equity = credit − debit
  1490→
  1491→    // ── Detailed item detection: prefix first, then keyword fallback ──
  1492→    // Inventory
  1493→    if (matchesPrefix(num, inventoryPrefix) || (!inventoryPrefix && hasKeyword(n, INVENTORY_KEYWORDS))) {
  1494→      inventory += debitBal;
  1495→    }
  1496→    // Cash & equivalents
  1497→    if (matchesPrefix(num, cashPrefix) || (!cashPrefix && hasKeyword(n, CASH_KEYWORDS))) {
  1498→      cash += debitBal;
  1499→    }
  1500→    // Receivables
  1501→    if (matchesPrefix(num, receivablesPrefix) || (!receivablesPrefix && hasKeyword(n, RECEIVABLES_KEYWORDS))) {
  1502→      receivables += debitBal;
  1503→    }
  1504→    // Fixed assets
  1505→    if (matchesPrefix(num, fixedAssetsPrefix) || (!fixedAssetsPrefix && hasKeyword(n, FIXED_ASSETS_KEYWORDS))) {
  1506→      fixedAssets += debitBal;
  1507→    }
  1508→    // Payables
  1509→    if (matchesPrefix(num, payablesPrefix) || (!payablesPrefix && hasKeyword(n, PAYABLES_KEYWORDS))) {
  1510→      payables += creditBal;
  1511→    }
  1512→    // Short-term debt
  1513→    if (matchesPrefix(num, shortTermDebtPrefix) || (!shortTermDebtPrefix && hasKeyword(n, SHORT_TERM_DEBT_KEYWORDS))) {
  1514→      shortTermDebt += creditBal;
  1515→    }
  1516→    // Long-term debt
  1517→    if (matchesPrefix(num, longTermDebtPrefix) || (!longTermDebtPrefix && hasKeyword(n, LONG_TERM_DEBT_KEYWORDS))) {
  1518→      longTermDebt += creditBal;
  1519→    }
  1520→
  1521→    // ── Main category detection by prefix ──
  1522→    if (matchesPrefix(num, currentAssetsPrefix)) {
  1523→      currentAssets += debitBal;
  1524→    } else if (matchesPrefix(num, nonCurrentAssetsPrefix)) {
  1525→      nonCurrentAssets += debitBal;
  1526→    } else if (matchesPrefix(num, currentLiabPrefix)) {
  1527→      currentLiabilities += creditBal;
  1528→    } else if (matchesPrefix(num, nonCurrentLiabPrefix)) {
  1529→      nonCurrentLiabilities += creditBal;
  1530→    } else if (matchesPrefix(num, equityPrefix)) {
  1531→      equity += creditBal;
  1532→    } else if (num.indexOf("1") === 0) {
  1533→      currentAssets += debitBal;
  1534→    } else if (num.indexOf("2") === 0) {
  1535→      currentLiabilities += creditBal;
  1536→    } else if (num.indexOf("3") === 0) {
  1537→      equity += creditBal;
  1538→    }
  1539→  }
  1540→
  1541→  const totalAssets = r2(currentAssets + nonCurrentAssets);
  1542→  const totalLiabilities = r2(currentLiabilities + nonCurrentLiabilities);
  1543→  const totalDebt = r2(shortTermDebt + longTermDebt);
  1544→
  1545→  return {
  1546→    currentAssets: r2(currentAssets),
  1547→    nonCurrentAssets: r2(nonCurrentAssets),
  1548→    totalAssets,
  1549→    currentLiabilities: r2(currentLiabilities),
  1550→    nonCurrentLiabilities: r2(nonCurrentLiabilities),
  1551→    totalLiabilities,
  1552→    equity: r2(equity),
  1553→    inventory: r2(inventory),
  1554→    cash: r2(cash),
  1555→    receivables: r2(receivables),
  1556→    fixedAssets: r2(fixedAssets),
  1557→    payables: r2(payables),
  1558→    shortTermDebt: r2(shortTermDebt),
  1559→    longTermDebt: r2(longTermDebt),
  1560→    totalDebt,
  1561→  };
  1562→}

  1601→/** Convert a turnover ratio (e.g. inventoryTurnover) to days (365 / turnover). Returns null if input is null/0.

  1816→/** Format a ratio value for display. */
  1817→export function fmtRatio(v: number | null, unit: "ratio" | "percent" | "amount" | "days"): string {
  1818→  if (v == null) return "—";
  1819→  if (unit === "percent") return (v * 100).toFixed(1) + "%";
  1820→  if (unit === "amount") return v.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  1821→  if (unit === "days") return v.toFixed(0) + " يوم";
  1822→  return v.toFixed(2) + "×";
  1823→}
