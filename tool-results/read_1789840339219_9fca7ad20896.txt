   118→function fmtKpi(v: number | null, format: "amount" | "ratio" | "percent"): string {
   119→  if (format === "amount") return fmtAmount(v);
   120→  if (format === "ratio") return v == null ? "—" : v.toFixed(2) + "×";
   121→  return fmtRatio(v, "percent");
   122→}

   124→/** شارة التغير: نسبة مئوية أو نقاط مئوية أو فرق مضاعف. */
   125→function DeltaChip({ delta, kind, positiveIsGood }: { delta: number | null; kind: "pct" | "pp" | "none"; positiveIsGood: boolean }) {
   126→  if (delta == null) return null;
   127→  const sign = delta > 0 ? "+" : "";
   128→  const text =
   129→    kind === "pct" ? `${sign}${(delta * 100).toFixed(1)}%`
   130→    : kind === "pp" ? `${sign}${(delta * 100).toFixed(1)} ن.م`
   131→    : `${sign}${delta.toFixed(2)}×`;
   132→  const good = delta === 0 ? null : delta > 0 === positiveIsGood;
   133→  return (
   134→    <span
   135→      className={cn(
   136→        "tnum inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[10px] font-bold",
   137→        good === null
   138→          ? "bg-slate-100 text-slate-500 dark:bg-slate-800 dark:text-slate-400"
   139→          : good
   140→            ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300"
   141→            : "bg-rose-100 text-rose-700 dark:bg-rose-900/50 dark:text-rose-300",
   142→      )}
   143→    >
   144→      {delta > 0 ? <ArrowUpRight className="size-3" /> : delta < 0 ? <ArrowDownRight className="size-3" /> : null}
   145→      {text}
   146→    </span>
   147→  );
   148→}

   236→/* ── استخراج أوراق الحسابات (قائمة الدخل) ───────────────────────────── */
   237→
   238→interface LeafItem {
   239→  num: string | null;
   240→  name: string;
   241→  v1: number | null;
   242→  v2: number | null;
   243→  chg: number | null;
   244→  cp: number | null;
   245→}
   246→
   247→function leafItems(rows: MatchedRow[], isRevenue: boolean): LeafItem[] {
   248→  const info = leafInfo(rows);
   249→  const items: LeafItem[] = [];
   250→  for (let i = 0; i < rows.length; i++) {
   251→    const { leaf1, leaf2 } = info[i];
   252→    if (!leaf1 && !leaf2) continue;
   253→    const r = rows[i];
   254→    const a = r.a1 ?? r.a2!;
   255→    const v1 = rowAmount(r, "a1", isRevenue);
   256→    const v2 = rowAmount(r, "a2", isRevenue);
   257→    const chg = v1 != null && v2 != null ? Math.round((v2 - v1) * 100) / 100 : null;
   258→    const cp = computeChangePct(chg, v1);
   259→    items.push({ num: a.num, name: a.nm, v1, v2, chg, cp });
   260→  }
   261→  items.sort((a, b) => Math.abs(b.v2 ?? 0) - Math.abs(a.v2 ?? 0));
   262→  return items;
   263→}

   542→function ChangeRankingChart({ cat }: { cat: Categorized }) {

   557→  collect(cat.cost, false);
   558→  collect(cat.sell, false);
   559→  collect(cat.adm, false);
   560→  collect(cat.oth, false);

   632→function MoversPanel({ movers }: { movers: MoverItem[] }) {
   633→  const best = movers
   634→    .filter((m) => m.favorable === true)
   635→    .sort((a, b) => (b.cp ?? Math.abs(b.chg)) - (a.cp ?? Math.abs(a.chg)))
   636→    .slice(0, 5);
   637→  const worst = movers
   638→    .filter((m) => m.favorable === false)
   639→    .sort((a, b) => Math.abs(b.cp ?? b.chg) - Math.abs(a.cp ?? a.chg))
   640→    .slice(0, 5);
   641→
   642→  if (best.length === 0 && worst.length === 0) return null;
   643→
   644→  const Row = ({ m, good }: { m: MoverItem; good: boolean }) => (
   645→    <div className="flex items-center justify-between gap-2 rounded-lg border border-slate-100 bg-white px-2.5 py-2 dark:border-slate-800 dark:bg-slate-900/50">

   648→          {m.num ? <span className="tnum me-1 text-[10px] text-slate-400">{m.num}</span> : null}

   650→        </p>
   651→        <p className="tnum text-[10px] text-slate-400 dark:text-slate-500">
   652→          {fmtAmount(m.v1)} ← {fmtAmount(m.v2)}
   653→        </p>
   654→      </div>
   655→      <div className="flex shrink-0 flex-col items-end gap-0.5">
   656→        <span className={cn(
   657→          "tnum text-[11px] font-bold",
   658→          good ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400",
   659→        )}>
   660→          {m.chg > 0 ? "+" : ""}{m.chg.toLocaleString("en-US")}

   662→        <span className={cn(
   663→          "tnum rounded-full px-1.5 text-[10px] font-bold",
   664→          good
   665→            ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/50 dark:text-emerald-300"

   682→      </CardHeader>
   683→      <CardContent className="grid max-h-96 gap-4 overflow-y-auto pe-1 sm:grid-cols-2">
   684→        <div className="space-y-1.5">

   687→          </p>

   689→            ? <p className="py-4 text-center text-[11px] text-slate-400">لا توجد بنود مرغوبة</p>

   915→function RatioGroupChart({
   916→  title, items, unit, L1, L2,
   917→}: {
   918→  title: string;

   927→    if (v == null) return "—";

   938→    comparative: scale(it.v1),

   941→  const cfg: ChartConfig = {
   942→    current: { label: L2, color: C.current },

   983→          {items.filter((it) => it.benchmark).map((it) => (
   984→            <span key={it.name} className="rounded-full bg-slate-100 px-2 py-0.5 text-[10px] text-slate-500 dark:bg-slate-800 dark:text-slate-400">
